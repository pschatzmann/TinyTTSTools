97enum class PhonemeModifier : uint8_t {
98 MOD_NONE = 0,
99
100 // -- Stress (vowels) ------------------------------------------------------
101 // Phonemes.h's `Phone` enum used to have Phone::AA1/AA2-style
102 // stress-variant ids (one pair per vowel, 30 ids total) purely as C++
103 // authoring sugar over the compact dictionary's `(baseId << 2) | stress`
104 // packing (see CompactPhonemeDictionaryBuilder.h) -- they've been removed
105 // (the packed-byte format itself is unchanged; a stressed word in
106 // PH_WORD() tables is now authored as Seg(Phone::X, MOD_STRESS_PRIMARY)
107 // instead). This value is the general-purpose replacement for any future
108 // non-English G2P built on Segment, so it never needs to double every
109 // vowel id the way AA1/AA2 did.
110 MOD_STRESS_PRIMARY = 1, // ˈ main stress of the word
111 MOD_STRESS_SECONDARY = 2, // ˌ secondary stress
112
113 // -- Length (applies mainly to vowels, also geminate consonants) --------
114 MOD_LONG = 3, // ː e.g. Japanese/Finnish/Arabic long vowels,
115 // Italian/Japanese geminate consonants
116 MOD_HALF_LONG = 4, // ˑ intermediate length (some Scandinavian,
117 // Estonian's 3-way quantity system)
118
119 // -- Nasalization (vowels/approximants) ----------------------------------
120 MOD_NASALIZED = 5, // ~ e.g. Portuguese/Polish/Hindi nasal vowels beyond
121 // the 4 French ones already given dedicated
122 // symbols (AN/EN/ON/UN)
123
124 // -- Secondary articulation (mainly consonants) --------------------------
125 MOD_PALATALIZED = 6, // ʲ e.g. Russian/Irish "soft" consonants
126 MOD_LABIALIZED = 7, // ʷ e.g. Arabic/Northwest Caucasian
127 MOD_VELARIZED = 8, // ˠ e.g. Russian/Irish "hard"/dark consonants
128 MOD_PHARYNGEALIZED = 9, // ˤ Arabic "emphatic" consonants (ص ض ط ظ)
129
130 // -- Phonation / voice onset time (mainly consonants) --------------------
131 MOD_ASPIRATED = 10, // ʰ phonemic in Mandarin/Korean/Thai/Hindi
132 // (3-4 way stop contrast, not just
133 // voiced/unvoiced)
134 MOD_UNRELEASED = 11, // ̚ e.g. Mandarin/Cantonese/Vietnamese
135 // syllable-final stops (no audible burst)
136 MOD_DEVOICED = 12, // ̥ partial devoicing of an otherwise-voiced
137 // segment (Japanese devoiced high vowels,
138 // German final-obstruent devoicing)
139 MOD_VOICED = 13, // ̬ allophonic voicing on a normally-voiceless
140 // segment (Spanish/Korean intervocalic
141 // lenition, English flapping)
142
143 // -- Phonation type (vowels, mainly) -------------------------------------
144 MOD_BREATHY = 14, // ̤ e.g. Gujarati, Hmong, some Burmese registers
145 MOD_CREAKY = 15, // ̰ e.g. Danish stød, Hmong, Vietnamese, Burmese
146
147 // -- Syllabicity / rhoticity ----------------------------------------------
148 MOD_SYLLABIC = 16, // ̩ a consonant filling a syllable nucleus
149 // (English "bottle" /bɒtl̩/, Czech "vlk")
150 MOD_RHOTACIZED = 17, // ˞ r-coloring added to a vowel (distinct from
151 // ARPAbet's dedicated ER, for languages where
152 // any vowel can be rhotacized, e.g. Mandarin
153 // erhua)
154
155 // -- Lexical tone (contour, on a syllable's vowel/nucleus) -----------------
156 // Formerly a separate ToneAccent.h (ToneShape/ToneSpec/Syllable),
157 // deleted as unused dead code (zero consumers) -- folded in here
158 // instead, matching how tone is conventionally written in practice
159 // (e.g. Mandarin Pinyin's tone diacritics sit directly ON the vowel,
160 // not as a wrapper around the whole syllable). Named descriptively
161 // (not "Mandarin tone 1-4") since the same shapes recur across
162 // unrelated tone languages with different pitch ranges/registers.
163 // Each value's pitch-contour effect (relative to whatever F0 the
164 // phoneme would otherwise use) is in deriveModifierEffect() below,
165 // consumed by FormantVocoder via PhonemeSynthesisParams::f0StartRatio/
166 // MidRatio/EndRatio -- the same 3-point interpolation mechanism
167 // ToneAccent.h's ToneSpec sketched, just reached through a modifier
168 // instead of a dedicated struct. PitchAccentPattern (Japanese-style
169 // per-mora high/low, a WORD-level structure, not reducible to a single
170 // segment's tag) had no such natural single-value form and was dropped
171 // along with the rest of that file, not migrated.
172 TONE_LEVEL_HIGH = 18, // ˥ e.g. Mandarin tone 1 (55), Cantonese tone 1
173 TONE_LEVEL_MID = 19, // ˧ e.g. Cantonese tone 3, Vietnamese "ngang"
174 TONE_LEVEL_LOW = 20, // ˩ e.g. Vietnamese "huyền"
175 TONE_RISING = 21, // e.g. Mandarin tone 2 (35), Vietnamese "sắc"
176 TONE_FALLING = 22, // e.g. Mandarin tone 4 (51), Vietnamese "nặng"
177 TONE_DIPPING = 23, // falling then rising, e.g. Mandarin tone 3 (214),
178 // Vietnamese "hỏi"
179 TONE_PEAKING = 24, // rising then falling within one syllable
180 TONE_CHECKED = 25, // short, glottalized, often stop-final -- e.g.
181 // Thai/Vietnamese "sắc"/"nặng" in closed
182 // syllables, Cantonese tones 1/3/6 in
183 // stop-final syllables
184};

278inline std::string parsePhonemeModifiers(const std::string& token,
279 PhonemeModifier& modifier) {
280 modifier = PhonemeModifier::MOD_NONE;
281 std::string s = token;
282 bool sawAny = false;
283
284 auto record = [&](PhonemeModifier found) {
285 if (sawAny && found != modifier) {
286 TTS_LOGW(
287 "parsePhonemeModifiers: '%s' carries more than one modifier tag "
288 "-- keeping the last one found, PhonemeModifier is "
289 "single-valued (see its own doc)",
290 token.c_str());
291 }
292 modifier = found;
293 sawAny = true;
294 };
295
296 bool changed = true;
297 while (changed) {
298 changed = false;
299 for (size_t i = 0; i < kModifierTagCount; ++i) {
300 const ModifierTag& t = kModifierTags[i];
301 if (t.position != ModifierPosition::PREFIX) continue;
302 size_t len = std::char_traits<char>::length(t.tag);
303 if (s.size() > len && s.compare(0, len, t.tag) == 0) {
304 record(t.bit);
305 s.erase(0, len);
306 changed = true;
307 }
308 }
309 }
310
311 changed = true;
312 while (changed) {
313 changed = false;
314 for (size_t i = 0; i < kModifierTagCount; ++i) {
315 const ModifierTag& t = kModifierTags[i];
316 if (t.position != ModifierPosition::SUFFIX) continue;
317 size_t len = std::char_traits<char>::length(t.tag);
318 if (s.size() > len && s.compare(s.size() - len, len, t.tag) == 0) {
319 record(t.bit);
320 s.erase(s.size() - len, len);
321 changed = true;
322 }
323 }
324 }
325
326 return s;
327}

339inline const ModifierTag* modifierTag(PhonemeModifier modifier) {
340 if (modifier == PhonemeModifier::MOD_NONE) return nullptr;
341 for (size_t i = 0; i < kModifierTagCount; ++i) {
342 if (kModifierTags[i].bit == modifier) return &kModifierTags[i];
343 }
344 return nullptr;
345}

355inline std::string applyModifierTag(const std::string& base, PhonemeModifier modifier) {
356 const ModifierTag* t = modifierTag(modifier);
357 if (t == nullptr) return base;
358 return t->position == ModifierPosition::PREFIX ? std::string(t->tag) + base
359 : base + t->tag;
360}

388inline ModifierEffect deriveModifierEffect(PhonemeModifier modifier) {
389 ModifierEffect e;
390 switch (modifier) {
391 case PhonemeModifier::MOD_STRESS_PRIMARY:
392 e.speedMul /= 1.2f;
393 e.volumeMul *= 1.15f;
394 break;
395 case PhonemeModifier::MOD_STRESS_SECONDARY:
396 e.speedMul /= 1.1f;
397 e.volumeMul *= 1.07f;
398 break;
399 case PhonemeModifier::MOD_LONG:
400 e.speedMul /= 1.8f;
401 break;
402 case PhonemeModifier::MOD_HALF_LONG:
403 e.speedMul /= 1.3f;
404 break;
405 case PhonemeModifier::MOD_BREATHY:
406 e.voicingMul *= 0.6f;
407 break;
408 case PhonemeModifier::MOD_DEVOICED:
409 e.voicingIsAbsolute = true;
410 e.voicingAbsolute = 0.0f;
411 break;
412 case PhonemeModifier::MOD_VOICED:
413 e.voicingIsAbsolute = true;
414 e.voicingAbsolute = 1.0f;
415 break;
416 // Tone contours: ratios are relative to whatever F0 the phoneme would
417 // otherwise use (a five-level/Chao-tone-letter-style reference scale
418 // -- a concrete language should scale/replace these, not assume they
419 // transfer as-is; absolute pitch range varies a lot even among tone
420 // languages). TONE_CHECKED additionally shortens the syllable, since
421 // checked tones are characteristically short/glottalized.
422 case PhonemeModifier::TONE_LEVEL_HIGH:
423 e.f0StartRatio = e.f0MidRatio = e.f0EndRatio = 1.30f;
424 break;
425 case PhonemeModifier::TONE_LEVEL_MID:
426 e.f0StartRatio = e.f0MidRatio = e.f0EndRatio = 1.00f;
427 break;
428 case PhonemeModifier::TONE_LEVEL_LOW:
429 e.f0StartRatio = e.f0MidRatio = e.f0EndRatio = 0.75f;
430 break;
431 case PhonemeModifier::TONE_RISING:
432 e.f0StartRatio = 0.85f;
433 e.f0MidRatio = 1.00f;
434 e.f0EndRatio = 1.30f;
435 break;
436 case PhonemeModifier::TONE_FALLING:
437 e.f0StartRatio = 1.30f;
438 e.f0MidRatio = 1.00f;
439 e.f0EndRatio = 0.70f;
440 break;
441 case PhonemeModifier::TONE_DIPPING:
442 e.f0StartRatio = 1.00f;
443 e.f0MidRatio = 0.65f;
444 e.f0EndRatio = 1.15f;
445 break;
446 case PhonemeModifier::TONE_PEAKING:
447 e.f0StartRatio = 0.90f;
448 e.f0MidRatio = 1.30f;
449 e.f0EndRatio = 0.90f;
450 break;
451 case PhonemeModifier::TONE_CHECKED:
452 e.f0StartRatio = 1.10f;
453 e.f0MidRatio = 1.05f;
454 e.f0EndRatio = 1.00f;
455 e.speedMul *= 1.3f; // >1.0 shortens (see PhonemeSynthesisParams::
456 // speed doc -- speed>1.0 is faster/shorter),
457 // matching a checked tone's characteristic
458 // short, glottalized syllable.
459 break;
460 default:
461 break; // no effect for this modifier (or NONE)
462 }
463 return e;
464}
