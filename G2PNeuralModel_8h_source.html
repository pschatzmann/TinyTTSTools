132 std::vector<int> predict(const std::string& word) const {
133 std::vector<float> h(hidden_dim_, 0.0f);
134 for (uint32_t cp : decodeUtf8(word)) {
135 gruStep(embRow(enc_emb_, graphemeIndexFor(language_, cp)), h, enc_w_ih_, enc_w_hh_, enc_b_ih_, enc_b_hh_);
136 }
137 gruStep(embRow(enc_emb_, 2 /* </s> */), h, enc_w_ih_, enc_w_hh_, enc_b_ih_, enc_b_hh_);
138
139 std::vector<int> out;
140 const float* dec = embRow(dec_emb_, 2 /* <s> */);
141 for (int step = 0; step < 20; step++) {
142 gruStep(dec, h, dec_w_ih_, dec_w_hh_, dec_b_ih_, dec_b_hh_);
143 int idx = argmaxLogits(h);
144 if (idx == 3 /* </s> */) break;
145 out.push_back(idx);
146 dec = embRow(dec_emb_, idx);
147 }
148 return out;
149 }

151 bool parseWeights(const uint8_t* buf, size_t len, size_t& pos) {
152 if (!readHeader(buf, len, pos)) return false;
153
154 enc_emb_ = readFloats(buf, len, pos, (size_t)num_graphemes_ * hidden_dim_);
155 dec_emb_ = readFloats(buf, len, pos, (size_t)num_dec_symbols_ * hidden_dim_);
156 if (!enc_emb_ || !dec_emb_) return false;
157
158 if (!readQuantized(buf, len, pos, &enc_w_ih_)) return false;
159 if (!readQuantized(buf, len, pos, &enc_w_hh_)) return false;
160 if (!readQuantized(buf, len, pos, &dec_w_ih_)) return false;
161 if (!readQuantized(buf, len, pos, &dec_w_hh_)) return false;
162
163 enc_b_ih_ = readFloats(buf, len, pos, 3 * (size_t)hidden_dim_);
164 enc_b_hh_ = readFloats(buf, len, pos, 3 * (size_t)hidden_dim_);
165 dec_b_ih_ = readFloats(buf, len, pos, 3 * (size_t)hidden_dim_);
166 dec_b_hh_ = readFloats(buf, len, pos, 3 * (size_t)hidden_dim_);
167 fc_w_ = readFloats(buf, len, pos, (size_t)num_phonemes_ * hidden_dim_);
168 fc_b_ = readFloats(buf, len, pos, (size_t)num_phonemes_);
169 if (!enc_b_ih_ || !enc_b_hh_ || !dec_b_ih_ || !dec_b_hh_ || !fc_w_ || !fc_b_) return false;
170
171 // The original format also stores a (symbol_id, tone) pair per output
172 // class, resolved against TinyTTS's own multi-lingual symbol table --
173 // not needed here since symbolForIndex() maps straight from output
174 // index to that language's phoneme table, but still present in the
175 // buffer and must be skipped to validate the buffer's total length.
176 if (pos + 2 * (size_t)num_phonemes_ > len) return false;
177 pos += 2 * (size_t)num_phonemes_;
178 return true;
179 }

181 bool readHeader(const uint8_t* buf, size_t len, size_t& pos) {
182 if (pos + 16 > len) return false;
183 int32_t hidden_dim, num_graphemes, num_dec_symbols, num_phonemes;
184 std::memcpy(&hidden_dim, buf + pos, 4);
185 std::memcpy(&num_graphemes, buf + pos + 4, 4);
186 std::memcpy(&num_dec_symbols, buf + pos + 8, 4);
187 std::memcpy(&num_phonemes, buf + pos + 12, 4);
188 pos += 16;
189 if (hidden_dim <= 0 || num_graphemes <= 0 || num_dec_symbols <= 0 || num_phonemes <= 0) return false;
190 hidden_dim_ = hidden_dim;
191 num_graphemes_ = num_graphemes;
192 num_dec_symbols_ = num_dec_symbols;
193 num_phonemes_ = num_phonemes;
194 return true;
195 }

197 static const float* readFloats(const uint8_t* buf, size_t len, size_t& pos, size_t count) {
198 size_t bytes = count * sizeof(float);
199 if (pos + bytes > len) return nullptr;
200 const float* p = reinterpret_cast<const float*>(buf + pos);
201 pos += bytes;
202 return p;
203 }

205 bool readQuantized(const uint8_t* buf, size_t len, size_t& pos, QuantizedWeight* w) const {
206 int rows = 3 * hidden_dim_;
207 size_t data_bytes = (size_t)rows * hidden_dim_;
208 if (pos + data_bytes > len) return false;
209 w->data = reinterpret_cast<const int8_t*>(buf + pos);
210 pos += data_bytes;
211 w->row_scale = readFloats(buf, len, pos, (size_t)rows);
212 w->rows = rows;
213 return w->row_scale != nullptr;
214 }

221 static std::vector<uint32_t> decodeUtf8(const std::string& s) {
222 std::vector<uint32_t> out;
223 size_t i = 0;
224 size_t n = s.size();
225 while (i < n) {
226 uint8_t c = (uint8_t)s[i];
227 if (c < 0x80) {
228 out.push_back(c);
229 i += 1;
230 } else if ((c & 0xE0) == 0xC0 && i + 1 < n) {
231 out.push_back(((uint32_t)(c & 0x1F) << 6) | ((uint8_t)s[i + 1] & 0x3F));
232 i += 2;
233 } else if ((c & 0xF0) == 0xE0 && i + 2 < n) {
234 out.push_back(((uint32_t)(c & 0x0F) << 12) | (((uint8_t)s[i + 1] & 0x3F) << 6) |
235 ((uint8_t)s[i + 2] & 0x3F));
236 i += 3;
237 } else if ((c & 0xF8) == 0xF0 && i + 3 < n) {
238 out.push_back(((uint32_t)(c & 0x07) << 18) | (((uint8_t)s[i + 1] & 0x3F) << 12) |
239 (((uint8_t)s[i + 2] & 0x3F) << 6) | ((uint8_t)s[i + 3] & 0x3F));
240 i += 4;
241 } else {
242 out.push_back(0xFFFD);
243 i += 1;
244 }
245 }
246 return out;
247 }

308 // in-place GRU cell: h is both the input hidden state and the output.
309 void gruStep(const float* x, std::vector<float>& h, const QuantizedWeight& w_ih, const QuantizedWeight& w_hh,
310 const float* b_ih, const float* b_hh) const {
311 int hidden = hidden_dim_;
312 std::vector<float> rzn_ih(3 * hidden), rzn_hh(3 * hidden);
313 // dequantized_weight[i][j] == row[j] * row_scale[i], so
314 // dot(x, dequantized_row) == row_scale[i] * dot(x, row) -- scale the
315 // int8 dot product once at the end rather than dequantizing every
316 // element first.
317 for (int i = 0; i < 3 * hidden; i++) {
318 const int8_t* row = w_ih.data + (size_t)i * hidden;
319 float dot = 0.0f;
320 for (int j = 0; j < hidden; j++) dot += x[j] * (float)row[j];
321 rzn_ih[i] = b_ih[i] + dot * w_ih.row_scale[i];
322 }
323 for (int i = 0; i < 3 * hidden; i++) {
324 const int8_t* row = w_hh.data + (size_t)i * hidden;
325 float dot = 0.0f;
326 for (int j = 0; j < hidden; j++) dot += h[j] * (float)row[j];
327 rzn_hh[i] = b_hh[i] + dot * w_hh.row_scale[i];
328 }
329 for (int i = 0; i < hidden; i++) {
330 float r = 1.0f / (1.0f + std::exp(-(rzn_ih[i] + rzn_hh[i])));
331 float z = 1.0f / (1.0f + std::exp(-(rzn_ih[hidden + i] + rzn_hh[hidden + i])));
332 float n = std::tanh(rzn_ih[2 * hidden + i] + r * rzn_hh[2 * hidden + i]);
333 h[i] = (1 - z) * n + z * h[i];
334 }
335 }

337 int argmaxLogits(const std::vector<float>& h) const {
338 int best = 0;
339 float best_val = -1e30f;
340 for (int j = 0; j < num_phonemes_; j++) {
341 float logit = fc_b_[j];
342 const float* row = fc_w_ + (size_t)j * hidden_dim_;
343 for (int k = 0; k < hidden_dim_; k++) logit += h[k] * row[k];
344 if (logit > best_val) {
345 best_val = logit;
346 best = j;
347 }
348 }
349 return best;
350 }
