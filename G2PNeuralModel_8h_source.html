101 std::vector<int> predict(const std::string& word) const {
102 std::vector<float> h(hidden_dim_, 0.0f);
103 for (char c : word) {
104 gruStep(embRow(enc_emb_, graphemeIndex(c)), h, enc_w_ih_, enc_w_hh_, enc_b_ih_, enc_b_hh_);
105 }
106 gruStep(embRow(enc_emb_, 2 /* </s> */), h, enc_w_ih_, enc_w_hh_, enc_b_ih_, enc_b_hh_);
107
108 std::vector<int> out;
109 const float* dec = embRow(dec_emb_, 2 /* <s> */);
110 for (int step = 0; step < 20; step++) {
111 gruStep(dec, h, dec_w_ih_, dec_w_hh_, dec_b_ih_, dec_b_hh_);
112 int idx = argmaxLogits(h);
113 if (idx == 3 /* </s> */) break;
114 out.push_back(idx);
115 dec = embRow(dec_emb_, idx);
116 }
117 return out;
118 }

120 bool parseWeights(const uint8_t* buf, size_t len, size_t& pos) {
121 if (!readHeader(buf, len, pos)) return false;
122
123 enc_emb_ = readFloats(buf, len, pos, (size_t)num_graphemes_ * hidden_dim_);
124 dec_emb_ = readFloats(buf, len, pos, (size_t)num_dec_symbols_ * hidden_dim_);
125 if (!enc_emb_ || !dec_emb_) return false;
126
127 if (!readQuantized(buf, len, pos, &enc_w_ih_)) return false;
128 if (!readQuantized(buf, len, pos, &enc_w_hh_)) return false;
129 if (!readQuantized(buf, len, pos, &dec_w_ih_)) return false;
130 if (!readQuantized(buf, len, pos, &dec_w_hh_)) return false;
131
132 enc_b_ih_ = readFloats(buf, len, pos, 3 * (size_t)hidden_dim_);
133 enc_b_hh_ = readFloats(buf, len, pos, 3 * (size_t)hidden_dim_);
134 dec_b_ih_ = readFloats(buf, len, pos, 3 * (size_t)hidden_dim_);
135 dec_b_hh_ = readFloats(buf, len, pos, 3 * (size_t)hidden_dim_);
136 fc_w_ = readFloats(buf, len, pos, (size_t)num_phonemes_ * hidden_dim_);
137 fc_b_ = readFloats(buf, len, pos, (size_t)num_phonemes_);
138 if (!enc_b_ih_ || !enc_b_hh_ || !dec_b_ih_ || !dec_b_hh_ || !fc_w_ || !fc_b_) return false;

150 bool readHeader(const uint8_t* buf, size_t len, size_t& pos) {
151 if (pos + 16 > len) return false;
152 int32_t hidden_dim, num_graphemes, num_dec_symbols, num_phonemes;
153 std::memcpy(&hidden_dim, buf + pos, 4);
154 std::memcpy(&num_graphemes, buf + pos + 4, 4);
155 std::memcpy(&num_dec_symbols, buf + pos + 8, 4);
156 std::memcpy(&num_phonemes, buf + pos + 12, 4);
157 pos += 16;
158 if (hidden_dim <= 0 || num_graphemes <= 0 || num_dec_symbols <= 0 || num_phonemes <= 0) return false;
159 hidden_dim_ = hidden_dim;
160 num_graphemes_ = num_graphemes;
161 num_dec_symbols_ = num_dec_symbols;
162 num_phonemes_ = num_phonemes;
163 return true;
164 }

166 static const float* readFloats(const uint8_t* buf, size_t len, size_t& pos, size_t count) {
167 size_t bytes = count * sizeof(float);
168 if (pos + bytes > len) return nullptr;
169 const float* p = reinterpret_cast<const float*>(buf + pos);
170 pos += bytes;
171 return p;
172 }

174 bool readQuantized(const uint8_t* buf, size_t len, size_t& pos, QuantizedWeight* w) const {
175 int rows = 3 * hidden_dim_;
176 size_t data_bytes = (size_t)rows * hidden_dim_;
177 if (pos + data_bytes > len) return false;
178 w->data = reinterpret_cast<const int8_t*>(buf + pos);
179 pos += data_bytes;
180 w->row_scale = readFloats(buf, len, pos, (size_t)rows);
181 w->rows = rows;
182 return w->row_scale != nullptr;
183 }

193 // in-place GRU cell: h is both the input hidden state and the output.
194 void gruStep(const float* x, std::vector<float>& h, const QuantizedWeight& w_ih, const QuantizedWeight& w_hh,
195 const float* b_ih, const float* b_hh) const {
196 int hidden = hidden_dim_;
197 std::vector<float> rzn_ih(3 * hidden), rzn_hh(3 * hidden);
198 // dequantized_weight[i][j] == row[j] * row_scale[i], so
199 // dot(x, dequantized_row) == row_scale[i] * dot(x, row) -- scale the
200 // int8 dot product once at the end rather than dequantizing every
201 // element first.
202 for (int i = 0; i < 3 * hidden; i++) {
203 const int8_t* row = w_ih.data + (size_t)i * hidden;
204 float dot = 0.0f;
205 for (int j = 0; j < hidden; j++) dot += x[j] * (float)row[j];
206 rzn_ih[i] = b_ih[i] + dot * w_ih.row_scale[i];
207 }
208 for (int i = 0; i < 3 * hidden; i++) {
209 const int8_t* row = w_hh.data + (size_t)i * hidden;
210 float dot = 0.0f;
211 for (int j = 0; j < hidden; j++) dot += h[j] * (float)row[j];
212 rzn_hh[i] = b_hh[i] + dot * w_hh.row_scale[i];
213 }
214 for (int i = 0; i < hidden; i++) {
215 float r = 1.0f / (1.0f + std::exp(-(rzn_ih[i] + rzn_hh[i])));
216 float z = 1.0f / (1.0f + std::exp(-(rzn_ih[hidden + i] + rzn_hh[hidden + i])));
217 float n = std::tanh(rzn_ih[2 * hidden + i] + r * rzn_hh[2 * hidden + i]);
218 h[i] = (1 - z) * n + z * h[i];
219 }
220 }

222 int argmaxLogits(const std::vector<float>& h) const {
223 int best = 0;
224 float best_val = -1e30f;
225 for (int j = 0; j < num_phonemes_; j++) {
226 float logit = fc_b_[j];
227 const float* row = fc_w_ + (size_t)j * hidden_dim_;
228 for (int k = 0; k < hidden_dim_; k++) logit += h[k] * row[k];
229 if (logit > best_val) {
230 best_val = logit;
231 best = j;
232 }
233 }
234 return best;
235 }
