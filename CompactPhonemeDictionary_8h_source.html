85 bool lookup(const std::string& word, std::string& outPhonemes) const override {
86 if (wordOffsets_ == nullptr || count_ == 0) return false;
87 int32_t lo = 0, hi = static_cast<int32_t>(count_) - 1;
88 while (lo <= hi) {
89 int32_t mid = lo + (hi - lo) / 2;
90 int cmp = compareWord(static_cast<uint32_t>(mid), word);
91 if (cmp == 0) {
92 decodePhonemes(static_cast<uint32_t>(mid), outPhonemes);
93 return true;
94 } else if (cmp < 0) {
95 lo = mid + 1;
96 } else {
97 hi = mid - 1;
98 }
99 }
100 return false;
101 }

127 int compareWord(uint32_t idx, const std::string& word) const {
128 uint32_t start = wordOffsets_[idx];
129 uint32_t end = wordOffsets_[idx + 1];
130 size_t wlen = end - start;
131 const char* w = wordsBlob_ + start;
132 size_t minLen = wlen < word.size() ? wlen : word.size();
133 int cmp = std::memcmp(w, word.data(), minLen);
134 if (cmp != 0) return cmp;
135 return static_cast<int>(wlen) - static_cast<int>(word.size());
136 }

138 void decodePhonemes(uint32_t idx, std::string& out) const {
139 out.clear();
140 uint32_t start = phonemeOffsets_[idx];
141 uint32_t end = phonemeOffsets_[idx + 1];
142 const uint8_t* p = phonemeData_ + start;
143 // One shared, lazily-constructed lookup table for id -> ARPABET string
144 // (function-local static: built once, not once per lookup/instance).
145 static const Phonemes phonemeTable;
146 for (uint32_t i = 0; i < end - start; ++i) {
147 uint8_t b = p[i];
148 uint8_t id = b >> 2;
149 uint8_t stress = b & 0x3;
150 const PhonemeInfo* info = phonemeTable.getPhonemeById(id);
151 if (!info) continue;
152 if (!out.empty()) out += ' ';
153 out += info->arpabet;
154 if (stress != 0) out += static_cast<char>('0' + stress);
155 }
156 }
