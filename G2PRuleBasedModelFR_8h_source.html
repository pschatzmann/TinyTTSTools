109 void applySingleLetter(const std::string& word, size_t i, std::string& result, size_t& advance) override {
110 char c = word[i];
111 char next = (i + 1 < word.length()) ? word[i + 1] : '\0';
112 bool wordEnd = (i + 1 == word.length());
113 advance = 1;
114
115 switch (c) {
116 case 'a': appendTokens(result, "AF"); break;
117 case 'e': appendTokens(result, wordEnd ? "" : "AH0"); break;
118 case 'i': appendTokens(result, "IY"); break;
119 case 'o': appendTokens(result, "AO"); break;
120 case 'u': appendTokens(result, "UF"); break;
121 case 'y': appendTokens(result, "IY"); break;
122 case 'b': appendTokens(result, "B"); break;
123 case 'c':
