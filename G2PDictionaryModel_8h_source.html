142
143 while (left <= right) {
144 int mid = left + (right - left) / 2;
145 int comparison = strcmp(word.c_str(), custom_dictionary_[mid].word);
146
147 if (comparison == 0) {
148 return std::string(custom_dictionary_[mid].phonemes);
