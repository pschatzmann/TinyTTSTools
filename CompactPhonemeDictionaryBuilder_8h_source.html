43constexpr bool cxWordLess(const char* a, const char* b) {
44 size_t i = 0;
45 while (a[i] != '\0' && b[i] != '\0') {
46 if (a[i] != b[i]) return static_cast<unsigned char>(a[i]) < static_cast<unsigned char>(b[i]);
47 ++i;
48 }
49 return a[i] == '\0' && b[i] != '\0';
50}

125constexpr uint16_t Seg(Phone base,

165template <size_t N, size_t WordBytes, size_t PhonemeCount>
166struct CompactPhonemeDictionaryData {
167 std::array<uint32_t, N + 1> wordOffsets{};
168 std::array<char, WordBytes> wordsBlob{};
169 std::array<uint32_t, N + 1> phonemeOffsets{};
170 std::array<uint16_t, PhonemeCount> phonemeData{};
171
172 constexpr CompactPhonemeDictionaryData(const PhonemeWordSource (&table)[N]) {
173 using namespace tts_compact_dict_detail;
174 size_t wPos = 0, pPos = 0;
175 wordOffsets[0] = 0;
176 phonemeOffsets[0] = 0;
177 for (size_t i = 0; i < N; ++i) {
178 if (i > 0 && !cxWordLess(table[i - 1].word, table[i].word)) {
179 throw "CompactPhonemeDictionaryData: source table is not sorted "
180 "(words must be strictly ascending for binary search)";
181 }
182 size_t wlen = cxStrLen(table[i].word);
183 for (size_t j = 0; j < wlen; ++j) wordsBlob[wPos + j] = table[i].word[j];
184 wPos += wlen;
185 wordOffsets[i + 1] = static_cast<uint32_t>(wPos);
186
187 for (size_t j = 0; j < table[i].phonemeCount; ++j) {
188 phonemeData[pPos++] = table[i].phonemeData[j];
189 }
190 phonemeOffsets[i + 1] = static_cast<uint32_t>(pPos);
191 }
192 }
193};
