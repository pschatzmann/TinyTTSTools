61constexpr uint8_t cxPackPhone(Phone p) {
62 uint8_t v = static_cast<uint8_t>(p);
63 if (v <= 42) return static_cast<uint8_t>(v << 2);
64 uint8_t idx = static_cast<uint8_t>(v - 43);
65 uint8_t baseId = kStressedVowelBaseIds[idx / 2];
66 uint8_t stress = static_cast<uint8_t>((idx % 2) + 1);
67 return static_cast<uint8_t>((baseId << 2) | stress);
68}

130struct CompactPhonemeDictionaryData {
131 std::array<uint32_t, N + 1> wordOffsets{};
132 std::array<char, WordBytes> wordsBlob{};

136 constexpr CompactPhonemeDictionaryData(const PhonemeWordSource (&table)[N]) {
137 using namespace tts_compact_dict_detail;
138 size_t wPos = 0, pPos = 0;
139 wordOffsets[0] = 0;
140 phonemeOffsets[0] = 0;
141 for (size_t i = 0; i < N; ++i) {
142 if (i > 0 && !cxWordLess(table[i - 1].word, table[i].word)) {
143 throw "CompactPhonemeDictionaryData: source table is not sorted "
144 "(words must be strictly ascending for binary search)";
145 }
146 size_t wlen = cxStrLen(table[i].word);
147 for (size_t j = 0; j < wlen; ++j) wordsBlob[wPos + j] = table[i].word[j];
148 wPos += wlen;
149 wordOffsets[i + 1] = static_cast<uint32_t>(wPos);
150
151 for (size_t j = 0; j < table[i].phonemeCount; ++j) {
152 phonemeData[pPos++] = table[i].phonemeData[j];
153 }
154 phonemeOffsets[i + 1] = static_cast<uint32_t>(pPos);
155 }
156 }
157};
