92 bool synthesizePhoneme(const std::string& phoneme, ::Print& out,
93 const PhonemeSynthesisParams& params) override {
94 std::vector<std::string> tokens = StringUtils::split(phoneme, ' ');
95 if (tokens.empty()) return false;
96
97 bool success = true;
98 for (const std::string& token : tokens) {
99 // Check silence against the bare (stress-stripped) name: audio
100 // dictionaries and Phonemes::isSilence() both key on bare phoneme
101 // names, and isSilence() treats any unrecognized symbol as silence
102 // -- without stripping first, every stressed vowel (e.g. "IH1")
103 // would silently turn to silence (see
104 // ConcatenatedAudioVocoder::processSequenceWithLookahead's own copy
105 // of this fix for the same reason). synthesizeOnePhoneme() does its
106 // own stripping internally too -- it still needs the original,
107 // stress-marked token to apply the stress-duration boost.
108 int stress;
109 std::string bare = stripStressMarker(token, stress);
110 if (translator_.isSilence(defaultPhonemeType, bare.c_str())) {
111 uint16_t durationMs = resolveDuration(
112 translator_.getPhonemeDuration(defaultPhonemeType, bare), params);
113 outputSilence(sampleRate(), durationMs, out);
114 } else if (!synthesizeOnePhoneme(token, out, params)) {
115 success = false;
116 }
117 }
118 return success;
119 }

125 bool synthesizeOnePhoneme(const std::string& phoneme, ::Print& out,
126 const PhonemeSynthesisParams& params) {
127 // Dictionaries are keyed by bare phoneme names (e.g. "AA", not "AA1") --
128 // strip the stress marker before lookup, same as the isSilence() check
129 // in synthesizePhoneme() above.
130 int stress;
131 std::string bare = stripStressMarker(phoneme, stress);
132
133 const SoundEntry* entry = dictionary_.getSoundEntry(bare.c_str());
134 if (!entry) return false;
135
136 size_t n = entry->samples();
137 if (n == 0) return false;
138
139 std::vector<float> src(n);
140 for (size_t i = 0; i < n; i++) src[i] = static_cast<float>((*entry)[i]);
141
142 uint16_t defaultDurationMs = getPhonemeWithStressDuration(bare, stress);
143 uint16_t targetDurationMs = resolveDuration(defaultDurationMs, params);
144 size_t targetSamples = (static_cast<size_t>(targetDurationMs) * sampleRate()) / 1000;
145 if (targetSamples == 0) targetSamples = n;
146
147 float targetHz = params.pitchHz > 0.0f ? params.pitchHz : defaultPitchHz_;
148 int synthPeriod = static_cast<int>(sampleRate() / targetHz);
149 if (synthPeriod < kMinPeriodSamples) synthPeriod = kMinPeriodSamples;
150
151 int sourcePeriod = estimatePitchPeriod(src, sampleRate());
152
153 std::vector<float> outBuf(targetSamples, 0.0f);
154 std::vector<float> weight(targetSamples, 0.0f);
155
156 double srcScale = static_cast<double>(n) / static_cast<double>(targetSamples);
157 int64_t windowLen = 2 * static_cast<int64_t>(sourcePeriod);
158
159 for (int64_t synthMark = 0; synthMark < static_cast<int64_t>(targetSamples);
160 synthMark += synthPeriod) {
161 int64_t rawCenter = static_cast<int64_t>(synthMark * srcScale + 0.5);
162 // Snap the extracted content to the nearest real source pitch-period
163 // boundary, reusing or skipping source periods as needed, instead of
164 // a raw proportional position: when target duration ~= source
165 // duration, srcScale ~= 1, so an unsnapped center tracks synthMark
166 // almost 1:1 regardless of the requested pitch, reconstructing the
167 // source's own original waveform near-verbatim and making pitchHz a
168 // no-op. Snapping decouples *what content* gets played (tied to
169 // real source periods) from *how densely* it's placed (tied to the
170 // target pitch), which is what actually lets overlap-add raise or
171 // lower the perceived rate.
172 int64_t analysisIdx = (rawCenter + sourcePeriod / 2) / sourcePeriod;
173 int64_t center = analysisIdx * sourcePeriod;
174 int64_t srcStart = center - sourcePeriod;
175 int64_t outStart = synthMark - sourcePeriod;
176
177 for (int64_t k = 0; k < windowLen; k++) {
178 int64_t srcIdx = srcStart + k;
179 int64_t outIdx = outStart + k;
180 if (srcIdx < 0 || srcIdx >= static_cast<int64_t>(n)) continue;
181 if (outIdx < 0 || outIdx >= static_cast<int64_t>(targetSamples)) continue;
182 // Hann window over the excerpt -- tapers each overlapping copy to
183 // zero at its edges so consecutive excerpts blend rather than
184 // click at their boundaries.
185 float hann = 0.5f - 0.5f * cosf(2.0f * M_PI * k / static_cast<float>(windowLen - 1));
186 outBuf[outIdx] += src[srcIdx] * hann;
187 weight[outIdx] += hann;
188 }
189 }
190
191 float volume = params.volume;
192 std::vector<int16_t> pcm(targetSamples);
193 for (size_t i = 0; i < targetSamples; i++) {
194 float v = weight[i] > 1e-6f ? outBuf[i] / weight[i] : 0.0f;
195 v *= volume;
196 if (v > 32767.0f) v = 32767.0f;
197 if (v < -32768.0f) v = -32768.0f;
198 pcm[i] = static_cast<int16_t>(v);
199 }
200
201 out.write(reinterpret_cast<const uint8_t*>(pcm.data()), pcm.size() * sizeof(int16_t));
202 return true;
203 }

205 static constexpr int kMinPeriodSamples = 8; // ~1000Hz ceiling at 8kHz, a sanity floor

210 int estimatePitchPeriod(const std::vector<float>& x, int sr) const {
211 int minLag = sr / 300;
212 int maxLag = sr / 60;
213 if (minLag < kMinPeriodSamples) minLag = kMinPeriodSamples;
214 if (maxLag >= static_cast<int>(x.size())) maxLag = static_cast<int>(x.size()) - 1;
215 if (minLag >= maxLag) return sr / 120; // unit too short to estimate -- fall back to ~120Hz
216
217 double bestScore = -1.0;
218 int bestLag = sr / 120;
219 for (int lag = minLag; lag <= maxLag; lag++) {
220 double sum = 0.0;
221 for (size_t i = 0; i + static_cast<size_t>(lag) < x.size(); i++) {
222 sum += static_cast<double>(x[i]) * x[i + lag];
223 }
224 if (sum > bestScore) {
225 bestScore = sum;
226 bestLag = lag;
227 }
228 }
229 return bestLag;
230 }
