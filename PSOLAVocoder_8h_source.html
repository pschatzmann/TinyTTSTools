136 bool synthesizePhoneme(const std::string& phoneme, ::Print& out,
137 const PhonemeSynthesisParams& params) override {
138 std::vector<std::string> tokens = StringUtils::split(phoneme, ' ');
139 if (tokens.empty()) return false;
140
141 bool success = true;
142 for (const std::string& token : tokens) {
143 // Check silence against the bare (stress- and modifier-stripped)
144 // name: audio dictionaries and Phonemes::isSilence() both key on
145 // bare phoneme names, and isSilence() treats any unrecognized
146 // symbol as silence -- without stripping first, every stressed
147 // vowel (e.g. "IH1") or modifier-tagged token (e.g. "AA:", "P_h")
148 // would silently turn to silence (see
149 // ConcatenatedAudioVocoder::processSequenceWithLookahead's own copy
150 // of this fix for the same reason). synthesizeOnePhoneme() does its
151 // own stripping internally too -- it still needs the original,
152 // stress/modifier-marked token to apply the stress-duration boost
153 // and modifier effects.
154 int stress;
155 std::string bare = stripStressMarker(token, stress);
156 PhonemeModifier unusedMod = PhonemeModifier::MOD_NONE;
157 bare = parsePhonemeModifiers(bare, unusedMod);
158 if (translator_.isSilence(defaultPhonemeType, bare.c_str())) {
159 uint16_t durationMs = resolveDuration(
160 translator_.getPhonemeDuration(defaultPhonemeType, bare), params);
161 outputSilence(sampleRate(), durationMs, out);
162 } else if (!synthesizeOnePhoneme(token, out, params)) {
163 success = false;
164 }
165 }
166 return success;
167 }

188 bool synthesizeOnePhoneme(const std::string& phoneme, ::Print& out,
189 const PhonemeSynthesisParams& params) {
190 // Dictionaries are keyed by bare phoneme names (e.g. "AA", not "AA1") --
191 // strip the stress marker before lookup, same as the isSilence() check
192 // in synthesizePhoneme() above.
193 int stress;
194 std::string bare = stripStressMarker(phoneme, stress);
195
196 // Modifier tag (e.g. "T_j"), if the token carries one, on top of the
197 // digit-stress convention above -- a PREFIX stress tag folds into the
198 // same `stress` int (so it drives getPhonemeWithStressDuration() the
199 // same way a digit-suffixed "IH1" would), everything else becomes the
200 // modifier this phoneme applies. Token-level wins over a caller-set
201 // params.modifier, matching FormantVocoder's own precedence.
202 PhonemeModifier tagMod = PhonemeModifier::MOD_NONE;
203 bare = parsePhonemeModifiers(bare, tagMod);
204 if (tagMod == PhonemeModifier::MOD_STRESS_PRIMARY) {
205 stress = 1;
206 tagMod = PhonemeModifier::MOD_NONE;
207 } else if (tagMod == PhonemeModifier::MOD_STRESS_SECONDARY) {
208 stress = 2;
209 tagMod = PhonemeModifier::MOD_NONE;
210 }
211 PhonemeModifier modifier =
212 tagMod != PhonemeModifier::MOD_NONE ? tagMod : params.modifier;
213 ModifierEffect modEffect = deriveModifierEffect(modifier);
214
215 const SoundEntry* entry = dictionary_.getSoundEntry(bare.c_str());
216 if (!entry) return false;
217
218 size_t n = entry->samples();
219 if (n == 0) return false;
220
221 std::vector<float> src(n);
222 for (size_t i = 0; i < n; i++) src[i] = static_cast<float>((*entry)[i]);
223
224 // MOD_ASPIRATED/MOD_SYLLABIC add a fixed extra chunk of time to the
225 // phoneme's own natural duration -- same amounts and rationale as
226 // FormantVocoder::preparePhonemeSynthesis() (a VOT gap / absorbed
227 // syllable-nucleus time), additive rather than a speed multiplier.
228 uint16_t defaultDurationMs = getPhonemeWithStressDuration(bare, stress);
229 if (modifier == PhonemeModifier::MOD_ASPIRATED) {
230 defaultDurationMs += 45;
231 } else if (modifier == PhonemeModifier::MOD_SYLLABIC) {
232 defaultDurationMs += 40;
233 }
234 // MOD_LONG/MOD_HALF_LONG/TONE_CHECKED (via modEffect.speedMul) stretch
235 // or shorten the syllable the same way a caller's own params.speed
236 // would -- composed multiplicatively with it, not overriding it.
237 PhonemeSynthesisParams effectiveParams = params;
238 effectiveParams.speed *= modEffect.speedMul;
239 uint16_t targetDurationMs = resolveDuration(defaultDurationMs, effectiveParams);
240 size_t targetSamples = (static_cast<size_t>(targetDurationMs) * sampleRate()) / 1000;
241 if (targetSamples == 0) targetSamples = n;
242
243 float targetHz = params.pitchHz > 0.0f ? params.pitchHz : defaultPitchHz_;
244 int sourcePeriod = estimatePitchPeriod(src, sampleRate());
245
246 std::vector<float> outBuf(targetSamples, 0.0f);
247 std::vector<float> weight(targetSamples, 0.0f);
248
249 double srcScale = static_cast<double>(n) / static_cast<double>(targetSamples);
250 int64_t windowLen = 2 * static_cast<int64_t>(sourcePeriod);
251
252 // Pitch contour (see PhonemeSynthesisParams::f0StartRatio/MidRatio/
253 // EndRatio, and PhonemeModifier's TONE_* values via modEffect):
254 // composed multiplicatively the same way FormantVocoder does. Unlike
255 // a procedural vocoder, PSOLA already places one synthesis mark per
256 // pitch period, so the contour is realized by computing each mark's
257 // OWN target period from its position-derived ratio, rather than one
258 // fixed synthPeriod for the whole phoneme.
259 float f0StartRatio = params.f0StartRatio * modEffect.f0StartRatio;
260 float f0MidRatio = params.f0MidRatio * modEffect.f0MidRatio;
261 float f0EndRatio = params.f0EndRatio * modEffect.f0EndRatio;
262
263 for (int64_t synthMark = 0; synthMark < static_cast<int64_t>(targetSamples);) {
264 float progress = targetSamples > 1
265 ? static_cast<float>(synthMark) /
266 static_cast<float>(targetSamples - 1)
267 : 0.0f;
268 float contourRatio =
269 progress < 0.5f
270 ? f0StartRatio + (f0MidRatio - f0StartRatio) * (progress * 2.0f)
271 : f0MidRatio + (f0EndRatio - f0MidRatio) * ((progress - 0.5f) * 2.0f);
272 int synthPeriod = static_cast<int>(sampleRate() / (targetHz * contourRatio));
273 if (synthPeriod < kMinPeriodSamples) synthPeriod = kMinPeriodSamples;
274
275 int64_t rawCenter = static_cast<int64_t>(synthMark * srcScale + 0.5);
276 // Snap the extracted content to the nearest real source pitch-period
277 // boundary, reusing or skipping source periods as needed, instead of
278 // a raw proportional position: when target duration ~= source
279 // duration, srcScale ~= 1, so an unsnapped center tracks synthMark
280 // almost 1:1 regardless of the requested pitch, reconstructing the
281 // source's own original waveform near-verbatim and making pitchHz a
282 // no-op. Snapping decouples *what content* gets played (tied to
283 // real source periods) from *how densely* it's placed (tied to the
284 // target pitch), which is what actually lets overlap-add raise or
285 // lower the perceived rate.
286 int64_t analysisIdx = (rawCenter + sourcePeriod / 2) / sourcePeriod;
287 int64_t center = analysisIdx * sourcePeriod;
288 int64_t srcStart = center - sourcePeriod;
289 int64_t outStart = synthMark - sourcePeriod;
290
291 for (int64_t k = 0; k < windowLen; k++) {
292 int64_t srcIdx = srcStart + k;
293 int64_t outIdx = outStart + k;
294 if (srcIdx < 0 || srcIdx >= static_cast<int64_t>(n)) continue;
295 if (outIdx < 0 || outIdx >= static_cast<int64_t>(targetSamples)) continue;
296 // Hann window over the excerpt -- tapers each overlapping copy to
297 // zero at its edges so consecutive excerpts blend rather than
298 // click at their boundaries.
299 float hann = 0.5f - 0.5f * cosf(2.0f * M_PI * k / static_cast<float>(windowLen - 1));
300 outBuf[outIdx] += src[srcIdx] * hann;
301 weight[outIdx] += hann;
302 }
303 synthMark += synthPeriod;
304 }
305
306 // Voicing (MOD_DEVOICED/MOD_BREATHY, and the pre-existing
307 // params.voicing this class didn't honor before): blend the
308 // PSOLA-reconstructed signal with generated broadband noise scaled to
309 // the same RMS, post-reconstruction -- see this class's own doc for
310 // why this is an approximation (and why MOD_VOICED is a no-op here).
311 float composedVoicing = params.voicing * modEffect.voicingMul;
312 if (modEffect.voicingIsAbsolute) composedVoicing = modEffect.voicingAbsolute;
313 if (composedVoicing < 0.0f) composedVoicing = 0.0f;
314 if (composedVoicing > 1.0f) composedVoicing = 1.0f;
315 if (composedVoicing < 0.999f) {
316 double sumSq = 0.0;
317 for (size_t i = 0; i < targetSamples; i++) {
318 float v = weight[i] > 1e-6f ? outBuf[i] / weight[i] : 0.0f;
319 sumSq += static_cast<double>(v) * v;
320 }
321 float rms = targetSamples > 0
322 ? sqrtf(static_cast<float>(sumSq / targetSamples))
323 : 0.0f;
324 for (size_t i = 0; i < targetSamples; i++) {
325 float v = weight[i] > 1e-6f ? outBuf[i] / weight[i] : 0.0f;
326 float noise = generateNoiseSource() * rms;
327 v = v * composedVoicing + noise * (1.0f - composedVoicing);
328 outBuf[i] = v;
329 weight[i] = 1.0f; // already blended -- normalize step below is a no-op
330 }
331 }
332
333 float volume = params.volume;
334 std::vector<int16_t> pcm(targetSamples);
335 for (size_t i = 0; i < targetSamples; i++) {
336 float v = weight[i] > 1e-6f ? outBuf[i] / weight[i] : 0.0f;
337 v *= volume;
338 if (v > 32767.0f) v = 32767.0f;
339 if (v < -32768.0f) v = -32768.0f;
340 pcm[i] = static_cast<int16_t>(v);
341 }
342
343 out.write(reinterpret_cast<const uint8_t*>(pcm.data()), pcm.size() * sizeof(int16_t));
344 return true;
345 }

352 int estimatePitchPeriod(const std::vector<float>& x, int sr) const {
353 int minLag = sr / 300;
354 int maxLag = sr / 60;
355 if (minLag < kMinPeriodSamples) minLag = kMinPeriodSamples;
356 if (maxLag >= static_cast<int>(x.size())) maxLag = static_cast<int>(x.size()) - 1;
357 if (minLag >= maxLag) return sr / 120; // unit too short to estimate -- fall back to ~120Hz
358
359 double bestScore = -1.0;
360 int bestLag = sr / 120;
361 for (int lag = minLag; lag <= maxLag; lag++) {
362 double sum = 0.0;
363 for (size_t i = 0; i + static_cast<size_t>(lag) < x.size(); i++) {
364 sum += static_cast<double>(x[i]) * x[i + lag];
365 }
366 if (sum > bestScore) {
367 bestScore = sum;
368 bestLag = lag;
369 }
370 }
371 return bestLag;
372 }
