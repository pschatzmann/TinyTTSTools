148 void addSampleToBatch(int16_t sample, ::Print& out, float volume = 1.0f) const {
149 if (volume != 1.0f) {
150 int32_t v = (int32_t)(sample * volume);
151 if (v > 32767) v = 32767;
152 else if (v < -32768) v = -32768;
153 sample = (int16_t)v;
154 }
155 const uint8_t* sampleBytes = reinterpret_cast<const uint8_t*>(&sample);
156 sampleBuffer_.insert(sampleBuffer_.end(), sampleBytes, sampleBytes + sizeof(int16_t));
157
158 // Check if we have enough bytes for the specified number of samples
159 if (sampleBuffer_.size() >= batchSize_ * sizeof(int16_t)) {
160 flushBatch(out);
161 }
162 }

186 int16_t sampleAtPitch(const SoundEntry* entry, size_t i, float pitchRatio) const {
187 if (pitchRatio == 1.0f) return (*entry)[i];
188 size_t maxIdx = entry->samples() > 0 ? entry->samples() - 1 : 0;
189 size_t srcIndex = static_cast<size_t>(i * pitchRatio);
190 if (srcIndex > maxIdx) srcIndex = maxIdx;
191 return (*entry)[srcIndex];
192 }

231 void processUnitCombination(const SoundEntry* currentEntry,
232 const SoundEntry* nextEntry,
233 uint16_t currentDuration,
234 ::Print& out,
235 const PhonemeSynthesisParams& params) {
236 if (!currentEntry) return;
237
238 size_t totalSamples = currentEntry->samples();
239 if (currentDuration > 0) {
240 size_t maxSamples = static_cast<size_t>((currentDuration * sampleRate()) / 1000);
241 if (maxSamples < totalSamples) totalSamples = maxSamples;
242 }
243
244 if (totalSamples == 0) return;
245
246 float pitchRatio = computePitchRatio(params.pitchHz);
247
248 // Track start offset if we apply cross-fade with previous unit
249 size_t startOffset = 0;
250
251 // Apply cross-fade with previous unit if we have stored tail
252 if (!lastUnitTail_.empty() && enableCrossFade_) {
253 // processCrossFadeWithPrevious() clears lastUnitTail_ before returning,
254 // so it must report how many samples it actually consumed -- reading
255 // lastUnitTail_.size() afterward (as done previously) always reads 0.
256 startOffset = processCrossFadeWithPrevious(currentEntry, totalSamples, out, params, pitchRatio);
257 }
258
259 // Determine processing parameters based on combination settings
260 size_t fadeOutStart = totalSamples;
261 size_t crossFadeStart = totalSamples;
262
263 if (nextEntry && enableCrossFade_) {
264 // Reserve samples for cross-fade
265 crossFadeStart = totalSamples > crossFadeSamples_ ?
266 totalSamples - crossFadeSamples_ : 0;
267 fadeOutStart = crossFadeStart; // No separate fade-out when cross-fading
268 } else if (totalSamples > fadeOutSamples_) {
269 // Not cross-fading into a next unit (disabled, or this is the last
270 // unit) -- always fade the ending out instead of cutting it abruptly.
271 fadeOutStart = totalSamples - fadeOutSamples_;
272 }
273
274 // Apply phase alignment for better cuts (end of current unit)
275 size_t finalSampleCount = applyPhaseAlignment(currentEntry, totalSamples, false);
276
277 // Process the main body of the unit (before fade/cross-fade)
278 size_t mainBodyEnd = std::min({fadeOutStart, crossFadeStart, finalSampleCount});
279 for (size_t i = startOffset; i < mainBodyEnd; i++) {
280 addSampleToBatch(sampleAtPitch(currentEntry, i, pitchRatio), out, params.volume);
281 }
282
283 // Handle ending: cross-fade or simple fade-out
284 if (nextEntry && enableCrossFade_ && crossFadeStart < finalSampleCount) {
285 storeUnitTailForCrossFade(currentEntry, crossFadeStart, finalSampleCount, nextEntry, pitchRatio);
286 } else if (fadeOutStart < finalSampleCount) {
287 processSimpleFadeOut(currentEntry, fadeOutStart, finalSampleCount, out, params.volume, pitchRatio);
288 lastUnitTail_.clear(); // Clear tail if not cross-fading
289 } else {
290 // No fading, write remaining samples
291 for (size_t i = mainBodyEnd; i < finalSampleCount; i++) {
292 addSampleToBatch(sampleAtPitch(currentEntry, i, pitchRatio), out, params.volume);
293 }
294 lastUnitTail_.clear(); // Clear tail if not cross-fading
295 }
296 }

305 size_t applyPhaseAlignment(const SoundEntry* entry, size_t totalSamples, bool isStart = false) {
306 // Degenerate case: "no adjustment" means start-at-0 for a start offset,
307 // but keep-the-full-length (totalSamples) for an end cutoff -- returning
308 // totalSamples for BOTH would make a start offset equal to the unit's
309 // own length, skipping its entire body.
310 if (totalSamples <= 1) return isStart ? 0 : totalSamples;
311
312 if (isStart) {
313 // For unit start: find zero crossing where amplitude exceeds stored threshold
314 return findOptimalStartPoint(entry, totalSamples);
315 } else {
316 // For unit end: find zero crossing and store negative amplitude
317 return findOptimalEndPoint(entry, totalSamples);
318 }
319 }

327 size_t findOptimalStartPoint(const SoundEntry* entry, size_t totalSamples) {
328 if (lastNegativeAmplitude_ == 0 || totalSamples <= 1) {
329 return 0; // No previous amplitude info, start at beginning
330 }
331
332 // Search first 25% of samples for appropriate zero crossing
333 size_t searchRange = std::min(totalSamples / 4, static_cast<size_t>(64)); // Limit search to 64 samples max
334
335 for (size_t i = 1; i < searchRange && i < totalSamples; i++) {
336 int16_t prevSample = (*entry)[i-1];
337 int16_t currSample = (*entry)[i];
338
339 // Look for zero upward crossing where amplitude exceeds threshold
340 if (prevSample <= 0 && currSample > 0) {
341 // Check if the positive amplitude exceeds our stored threshold
342 if (currSample >= lastNegativeAmplitude_) {
343 return i; // Start at this zero crossing
344 }
345 }
346 }
347
348 // If no suitable crossing found, use a simpler approach
349 for (size_t i = 1; i < searchRange && i < totalSamples; i++) {
350 if ((*entry)[i-1] <= 0 && (*entry)[i] > 0) {
351 return i; // Any zero upward crossing
352 }
353 }
354
355 return 0; // Default to start
356 }

364 size_t findOptimalEndPoint(const SoundEntry* entry, size_t totalSamples) {
365 // Search last 25% of samples for zero crossing
366 size_t searchRange = totalSamples / 4;
367 size_t searchStart = searchRange > 0 && searchRange < totalSamples ?
368 totalSamples - searchRange : 1;
369
370 // Track the largest negative amplitude in the end section
371 int16_t maxNegativeAmplitude = 0;
372 size_t optimalCutPoint = totalSamples;
373
374 for (size_t i = totalSamples - 1; i >= searchStart; --i) {
375 int16_t sample = (*entry)[i];
376
377 // Track negative amplitudes (store as positive values)
378 if (sample < 0) {
379 int16_t negativeAbs = static_cast<int16_t>(-sample);
380 if (negativeAbs > maxNegativeAmplitude) {
381 maxNegativeAmplitude = negativeAbs;
382 }
383 }
384
385 // Look for zero upward crossing
386 if (i > 0 && (*entry)[i-1] <= 0 && (*entry)[i] > 0) {
387 optimalCutPoint = i + 1; // Cut just after zero crossing
388 break;
389 }
390
391 if (i == searchStart) break; // Prevent underflow
392 }
393
394 // Store the maximum negative amplitude for next unit alignment
395 if (maxNegativeAmplitude > 0) {
396 lastNegativeAmplitude_ = maxNegativeAmplitude;
397 }
398
399 return optimalCutPoint;
400 }

414 size_t processCrossFadeWithPrevious(const SoundEntry* currentEntry, size_t totalSamples, ::Print& out,
415 const PhonemeSynthesisParams& params, float pitchRatio) {
416 // Find optimal start point for current unit using stored amplitude
417 size_t startOffset = applyPhaseAlignment(currentEntry, totalSamples, true);
418
419 // Adjust cross-fade parameters based on start offset
420 size_t availableSamples = totalSamples - startOffset;
421 size_t fadeSamples = std::min({crossFadeSamples_, lastUnitTail_.size(), availableSamples});
422
423 for (size_t i = 0; i < fadeSamples; i++) {
424 float fadeRatio = static_cast<float>(i) / static_cast<float>(fadeSamples);
425
426 // Previous unit sample (fading out) -- already pitch-shifted (if any)
427 // when it was captured by storeUnitTailForCrossFade()
428 int16_t prevSample = lastUnitTail_[i];
429 float prevWeight = 1.0f - fadeRatio;
430
431 // Current unit sample (fading in) - start from optimal offset
432 int16_t currentSample = sampleAtPitch(currentEntry, startOffset + i, pitchRatio);
433 float currentWeight = fadeRatio;
434
435 // Combine samples with cross-fade
436 int16_t combinedSample = static_cast<int16_t>(
437 (prevSample * prevWeight) + (currentSample * currentWeight)
438 );
439
440 addSampleToBatch(combinedSample, out, params.volume);
441 }
442
443 lastUnitTail_.clear();
444 return startOffset + fadeSamples;
445 }

455 void storeUnitTailForCrossFade(const SoundEntry* currentEntry, size_t fadeStart,
456 size_t fadeEnd, const SoundEntry* nextEntry, float pitchRatio) {
457 size_t fadeSamples = fadeEnd - fadeStart;
458
459 // Store the tail for cross-fade with next unit
460 lastUnitTail_.clear();
461 lastUnitTail_.reserve(fadeSamples);
462
463 for (size_t i = 0; i < fadeSamples; i++) {
464 int16_t sample = sampleAtPitch(currentEntry, fadeStart + i, pitchRatio);
465 lastUnitTail_.push_back(sample);
466 }
467 }

478 void processSimpleFadeOut(const SoundEntry* entry, size_t fadeStart, size_t fadeEnd, ::Print& out,
479 float volume, float pitchRatio) {
480 size_t fadeSamples = fadeEnd - fadeStart;
481
482 for (size_t i = 0; i < fadeSamples; i++) {
483 int16_t sample = sampleAtPitch(entry, fadeStart + i, pitchRatio);
484 float fadeMultiplier = 1.0f - (static_cast<float>(i) / static_cast<float>(fadeSamples));
485 sample = static_cast<int16_t>(sample * fadeMultiplier);
486 addSampleToBatch(sample, out, volume);
487 }

515 virtual bool processSequenceWithLookahead(const std::string& sequence, ::Print& out,
516 const PhonemeSynthesisParams& params) {
517 if (!isReady()) {
518 return false;
519 }
520
521 // Parse unit sequence
522 std::vector<std::string> units = StringUtils::split(sequence, ' ');
523 if (units.empty()) {
524 return false;
525 }
526
527 // Strip stress digits (e.g. "IH1" -> "IH") before any lookup: audio
528 // dictionaries are keyed by bare phoneme/diphone names with no stress
529 // marker, but a G2P source using the full CMU dictionary
530 // (COMPACT_CMUDICT_EN) emits stress-marked phonemes -- without this,
531 // every stressed vowel would silently fail every dictionary lookup.
532 for (std::string& unit : units) {
533 int stress;
534 unit = stripStressMarker(unit, stress);
535 }
536
537 bool success = true;
538
539 // Process each unit with context from the next unit
540 for (size_t i = 0; i < units.size(); i++) {
541 const std::string& currentUnit = units[i];
542
543 // Check for silence
544 if (phonemes_.isSilence(getDefaultPhonemeType(), currentUnit.c_str())) {
545 uint16_t duration = resolveDuration(
546 phonemes_.getPhonemeDuration(getDefaultPhonemeType(), currentUnit), params);
547 outputSilence(sampleRate(), duration, out);
548 continue;
549 }
550
551 // Get current unit entry
552 const SoundEntry* currentEntry = getAudioEntry(currentUnit);
553 if (!currentEntry) {
554 TTS_LOGW("ConcatenatedAudioVocoder: no audio entry for '%s' -- "
555 "dropped from output", currentUnit.c_str());
556 success = false;
557 continue;
558 }
559
560 // Get next unit entry for enhanced concatenation
561 const SoundEntry* nextEntry = nullptr;
562 if (i + 1 < units.size()) {
563 const std::string& nextUnit = units[i + 1];
564 if (!phonemes_.isSilence(getDefaultPhonemeType(), nextUnit.c_str())) {
565 nextEntry = getNextAudioEntry(currentUnit, nextUnit);
566 }
567 }
568
569 // Get duration and process with enhanced combination
570 uint16_t duration = resolveDuration(
571 phonemes_.getPhonemeDuration(getDefaultPhonemeType(), currentUnit), params);
572 processAudioUnit(currentEntry, nextEntry, duration, out, params);
573 }
574
575 return success;
576 }
