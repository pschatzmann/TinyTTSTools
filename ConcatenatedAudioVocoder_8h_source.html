170 void addSampleToBatch(int16_t sample, ::Print& out, float volume = 1.0f) const {
171 if (volume != 1.0f) {
172 int32_t v = (int32_t)(sample * volume);
173 if (v > 32767) v = 32767;
174 else if (v < -32768) v = -32768;
175 sample = (int16_t)v;
176 }
177 const uint8_t* sampleBytes = reinterpret_cast<const uint8_t*>(&sample);
178 sampleBuffer_.insert(sampleBuffer_.end(), sampleBytes, sampleBytes + sizeof(int16_t));
179
180 // Check if we have enough bytes for the specified number of samples
181 if (sampleBuffer_.size() >= batchSize_ * sizeof(int16_t)) {
182 flushBatch(out);
183 }
184 }

208 int16_t sampleAtPitch(const SoundEntry* entry, size_t i, float pitchRatio) const {
209 if (pitchRatio == 1.0f) return (*entry)[i];
210 size_t maxIdx = entry->samples() > 0 ? entry->samples() - 1 : 0;
211 size_t srcIndex = static_cast<size_t>(i * pitchRatio);
212 if (srcIndex > maxIdx) srcIndex = maxIdx;
213 return (*entry)[srcIndex];

253 void processUnitCombination(const SoundEntry* currentEntry,
254 const SoundEntry* nextEntry,
255 uint16_t currentDuration,
256 ::Print& out,
257 const PhonemeSynthesisParams& params) {
258 if (!currentEntry) return;
259
260 size_t totalSamples = currentEntry->samples();
261 if (currentDuration > 0) {
262 size_t maxSamples = static_cast<size_t>((currentDuration * sampleRate()) / 1000);
263 if (maxSamples < totalSamples) totalSamples = maxSamples;
264 }
265
266 if (totalSamples == 0) return;
267
268 float pitchRatio = computePitchRatio(params.pitchHz);
269
270 // Track start offset if we apply cross-fade with previous unit
271 size_t startOffset = 0;
272
273 // Apply cross-fade with previous unit if we have stored tail
274 if (!lastUnitTail_.empty() && enableCrossFade_) {
275 // processCrossFadeWithPrevious() clears lastUnitTail_ before returning,
276 // so it must report how many samples it actually consumed -- reading
277 // lastUnitTail_.size() afterward (as done previously) always reads 0.
278 startOffset = processCrossFadeWithPrevious(currentEntry, totalSamples, out, params, pitchRatio);
279 }
280
281 // Determine processing parameters based on combination settings
282 size_t fadeOutStart = totalSamples;
283 size_t crossFadeStart = totalSamples;
284
285 if (nextEntry && enableCrossFade_) {
286 // Reserve samples for cross-fade
287 crossFadeStart = totalSamples > crossFadeSamples_ ?
288 totalSamples - crossFadeSamples_ : 0;
289 fadeOutStart = crossFadeStart; // No separate fade-out when cross-fading
290 } else if (totalSamples > fadeOutSamples_) {
291 // Not cross-fading into a next unit (disabled, or this is the last
292 // unit) -- always fade the ending out instead of cutting it abruptly.
293 fadeOutStart = totalSamples - fadeOutSamples_;
294 }
295
296 // Apply phase alignment for better cuts (end of current unit)
297 size_t finalSampleCount = applyPhaseAlignment(currentEntry, totalSamples, false);
298
299 // Process the main body of the unit (before fade/cross-fade)
300 size_t mainBodyEnd = std::min({fadeOutStart, crossFadeStart, finalSampleCount});
301 for (size_t i = startOffset; i < mainBodyEnd; i++) {
302 addSampleToBatch(sampleAtPitch(currentEntry, i, pitchRatio), out, params.volume);
303 }
304
305 // Handle ending: cross-fade or simple fade-out
306 if (nextEntry && enableCrossFade_ && crossFadeStart < finalSampleCount) {
307 storeUnitTailForCrossFade(currentEntry, crossFadeStart, finalSampleCount, nextEntry, pitchRatio);
308 } else if (fadeOutStart < finalSampleCount) {
309 processSimpleFadeOut(currentEntry, fadeOutStart, finalSampleCount, out, params.volume, pitchRatio);
310 lastUnitTail_.clear(); // Clear tail if not cross-fading
311 } else {
312 // No fading, write remaining samples
313 for (size_t i = mainBodyEnd; i < finalSampleCount; i++) {
314 addSampleToBatch(sampleAtPitch(currentEntry, i, pitchRatio), out, params.volume);
315 }
316 lastUnitTail_.clear(); // Clear tail if not cross-fading
317 }
318 }

327 size_t applyPhaseAlignment(const SoundEntry* entry, size_t totalSamples, bool isStart = false) {
328 // Degenerate case: "no adjustment" means start-at-0 for a start offset,
329 // but keep-the-full-length (totalSamples) for an end cutoff -- returning
330 // totalSamples for BOTH would make a start offset equal to the unit's
331 // own length, skipping its entire body.
332 if (totalSamples <= 1) return isStart ? 0 : totalSamples;
333
334 if (isStart) {
335 // For unit start: find zero crossing where amplitude exceeds stored threshold
336 return findOptimalStartPoint(entry, totalSamples);
337 } else {
338 // For unit end: find zero crossing and store negative amplitude
339 return findOptimalEndPoint(entry, totalSamples);
340 }
341 }

349 size_t findOptimalStartPoint(const SoundEntry* entry, size_t totalSamples) {
350 if (lastNegativeAmplitude_ == 0 || totalSamples <= 1) {
351 return 0; // No previous amplitude info, start at beginning
352 }
353
354 // Search first 25% of samples for appropriate zero crossing
355 size_t searchRange = std::min(totalSamples / 4, static_cast<size_t>(64)); // Limit search to 64 samples max
356
357 for (size_t i = 1; i < searchRange && i < totalSamples; i++) {
358 int16_t prevSample = (*entry)[i-1];
359 int16_t currSample = (*entry)[i];
360
361 // Look for zero upward crossing where amplitude exceeds threshold
362 if (prevSample <= 0 && currSample > 0) {
363 // Check if the positive amplitude exceeds our stored threshold
364 if (currSample >= lastNegativeAmplitude_) {
365 return i; // Start at this zero crossing
366 }
367 }
368 }
369
370 // If no suitable crossing found, use a simpler approach
371 for (size_t i = 1; i < searchRange && i < totalSamples; i++) {
372 if ((*entry)[i-1] <= 0 && (*entry)[i] > 0) {
373 return i; // Any zero upward crossing
374 }
375 }
376
377 return 0; // Default to start
378 }

386 size_t findOptimalEndPoint(const SoundEntry* entry, size_t totalSamples) {
387 // Search last 25% of samples for zero crossing
388 size_t searchRange = totalSamples / 4;
389 size_t searchStart = searchRange > 0 && searchRange < totalSamples ?
390 totalSamples - searchRange : 1;
391
392 // Track the largest negative amplitude in the end section
393 int16_t maxNegativeAmplitude = 0;
394 size_t optimalCutPoint = totalSamples;
395
396 for (size_t i = totalSamples - 1; i >= searchStart; --i) {
397 int16_t sample = (*entry)[i];
398
399 // Track negative amplitudes (store as positive values)
400 if (sample < 0) {
401 int16_t negativeAbs = static_cast<int16_t>(-sample);
402 if (negativeAbs > maxNegativeAmplitude) {
403 maxNegativeAmplitude = negativeAbs;
404 }
405 }
406
407 // Look for zero upward crossing
408 if (i > 0 && (*entry)[i-1] <= 0 && (*entry)[i] > 0) {
409 optimalCutPoint = i + 1; // Cut just after zero crossing
410 break;
411 }
412
413 if (i == searchStart) break; // Prevent underflow
414 }
415
416 // Store the maximum negative amplitude for next unit alignment
417 if (maxNegativeAmplitude > 0) {
418 lastNegativeAmplitude_ = maxNegativeAmplitude;
419 }
420
421 return optimalCutPoint;
422 }

436 size_t processCrossFadeWithPrevious(const SoundEntry* currentEntry, size_t totalSamples, ::Print& out,
437 const PhonemeSynthesisParams& params, float pitchRatio) {
438 // Find optimal start point for current unit using stored amplitude
439 size_t startOffset = applyPhaseAlignment(currentEntry, totalSamples, true);
440
441 // Adjust cross-fade parameters based on start offset
442 size_t availableSamples = totalSamples - startOffset;
443 size_t fadeSamples = std::min({crossFadeSamples_, lastUnitTail_.size(), availableSamples});
444
445 for (size_t i = 0; i < fadeSamples; i++) {
446 float fadeRatio = static_cast<float>(i) / static_cast<float>(fadeSamples);
447
448 // Previous unit sample (fading out) -- already pitch-shifted (if any)
449 // when it was captured by storeUnitTailForCrossFade()
450 int16_t prevSample = lastUnitTail_[i];
451 float prevWeight = 1.0f - fadeRatio;
452
453 // Current unit sample (fading in) - start from optimal offset
454 int16_t currentSample = sampleAtPitch(currentEntry, startOffset + i, pitchRatio);
455 float currentWeight = fadeRatio;
456
457 // Combine samples with cross-fade
458 int16_t combinedSample = static_cast<int16_t>(
459 (prevSample * prevWeight) + (currentSample * currentWeight)
460 );
461
462 addSampleToBatch(combinedSample, out, params.volume);
463 }
464
465 lastUnitTail_.clear();
466 return startOffset + fadeSamples;
467 }

477 void storeUnitTailForCrossFade(const SoundEntry* currentEntry, size_t fadeStart,
478 size_t fadeEnd, const SoundEntry* nextEntry, float pitchRatio) {
479 size_t fadeSamples = fadeEnd - fadeStart;
480
481 // Store the tail for cross-fade with next unit
482 lastUnitTail_.clear();
483 lastUnitTail_.reserve(fadeSamples);
484
485 for (size_t i = 0; i < fadeSamples; i++) {
486 int16_t sample = sampleAtPitch(currentEntry, fadeStart + i, pitchRatio);
487 lastUnitTail_.push_back(sample);
488 }
489 }

500 void processSimpleFadeOut(const SoundEntry* entry, size_t fadeStart, size_t fadeEnd, ::Print& out,
501 float volume, float pitchRatio) {
502 size_t fadeSamples = fadeEnd - fadeStart;
503
504 for (size_t i = 0; i < fadeSamples; i++) {
505 int16_t sample = sampleAtPitch(entry, fadeStart + i, pitchRatio);
506 float fadeMultiplier = 1.0f - (static_cast<float>(i) / static_cast<float>(fadeSamples));
507 sample = static_cast<int16_t>(sample * fadeMultiplier);
508 addSampleToBatch(sample, out, volume);
509 }

537 virtual bool processSequenceWithLookahead(const std::string& sequence, ::Print& out,
538 const PhonemeSynthesisParams& params) {
539 if (!isReady()) {
540 return false;
541 }
542
543 // Parse unit sequence
544 std::vector<std::string> units = StringUtils::split(sequence, ' ');
545 if (units.empty()) {
546 return false;
547 }
548
549 // Strip stress digits (e.g. "IH1" -> "IH") before any lookup: audio
550 // dictionaries are keyed by bare phoneme/diphone names with no stress
551 // marker, but a G2P source using the full CMU dictionary
552 // (COMPACT_CMUDICT_EN) emits stress-marked phonemes -- without this,
553 // every stressed vowel would silently fail every dictionary lookup.
554 for (std::string& unit : units) {
555 int stress;
556 unit = stripStressMarker(unit, stress);
557 }
558
559 bool success = true;
560
561 // Process each unit with context from the next unit
562 for (size_t i = 0; i < units.size(); i++) {
563 const std::string& currentUnit = units[i];
564
565 // Check for silence
566 if (phonemes_.isSilence(getDefaultPhonemeType(), currentUnit.c_str())) {
567 uint16_t duration = resolveDuration(
568 phonemes_.getPhonemeDuration(getDefaultPhonemeType(), currentUnit), params);
569 outputSilence(sampleRate(), duration, out);
570 continue;
571 }
572
573 // Get current unit entry
574 const SoundEntry* currentEntry = getAudioEntry(currentUnit);
575 if (!currentEntry) {
576 TTS_LOGW("ConcatenatedAudioVocoder: no audio entry for '%s' -- "
577 "dropped from output", currentUnit.c_str());
578 success = false;
579 continue;
580 }
581
582 // Get next unit entry for enhanced concatenation
583 const SoundEntry* nextEntry = nullptr;
584 if (i + 1 < units.size()) {
585 const std::string& nextUnit = units[i + 1];
586 if (!phonemes_.isSilence(getDefaultPhonemeType(), nextUnit.c_str())) {
587 nextEntry = getNextAudioEntry(currentUnit, nextUnit);
588 }
589 }
590
591 // Get duration and process with enhanced combination
592 uint16_t duration = resolveDuration(
593 phonemes_.getPhonemeDuration(getDefaultPhonemeType(), currentUnit), params);
594 processAudioUnit(currentEntry, nextEntry, duration, out, params);
595 }
596
597 return success;
598 }
