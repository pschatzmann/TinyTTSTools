35// Desktop test-build convenience: example sketches are written against a

37// hardware) we transparently alias I2SStream to AudioTools' desktop
38// MiniAudioStream backend instead -- the sketch source needs no change.

40// desktop CMake example targets); real Arduino/embedded builds are
41// unaffected, and I2SStream itself is never defined by AudioTools on
42// desktop (it requires USE_I2S/IS_I2S_IMPLEMENTED), so this alias never

134 std::vector<std::string> toPhonemes(const std::string& text) {
135 // Step 1: Text preprocessing
136 std::vector<std::string> words = Tokenizer::tokenize(text);
137 if (words.empty()) {
138 if (errorCallback_) {
139 errorCallback_("Empty text input");
140 }
141 return {};
142 }
143
144 // Step 2: Grapheme-to-phoneme conversion
145 std::vector<std::string> phonemes;
146 for (const std::string& word : words) {
147 if (word == ".") {
148 phonemes.push_back("SIL"); // Add silence for punctuation
149 } else {
150 std::string wordPhonemes = g2p_->wordToPhonemes(word);
151 std::vector<std::string> wordPhonemeList =
152 StringUtils::split(wordPhonemes, ' ');
153 phonemes.insert(phonemes.end(), wordPhonemeList.begin(),
154 wordPhonemeList.end());
155 phonemes.push_back("SP"); // Add short pause between words
156 }
157 }
158
159 return phonemes;
160 }

187 bool say(const std::string& text) {
188 if (!initialized_) {
189 if (errorCallback_) {
190 errorCallback_("TTS not initialized");
191 }
192 return false;
193 }
194
195 // Step 1: Convert text to phonemes
196 std::vector<std::string> phonemes = toPhonemes(text);
197 if (phonemes.empty()) {
198 return false;
199 }
200
201 return sayPhonemes(g2p_->getDefaultPhonemeType(), phonemes);
202 }

213 bool sayPhonemes(PhonemeType phonemeType,
214 const std::vector<std::string>& phonemes) {
215 if (!initialized_) {
216 if (errorCallback_) {
217 errorCallback_("TTS not initialized");
218 }
219 return false;
220 }
221
222 if (phonemes.empty()) {
223 if (errorCallback_) {
224 errorCallback_("Empty phoneme sequence");
225 }
226 return false;
227 }
228
229 // Join into ONE space-separated sequence and make a single call, rather
230 // than calling sayPhoneme() once per individual phoneme token. Every
231 // vocoder that needs multi-phoneme context (DiphoneVocoder to form any
232 // diphone pair at all; PhonemeVocoder for cross-unit lookahead/
233 // coarticulation) only gets that context from the phonemes it sees
234 // *within one call* -- calling one phoneme at a time silently starved
235 // them of it: DiphoneVocoder in particular can't form a diphone at all
236 // from a single phoneme, so it fell back to treating every phoneme as
237 // its own isolated one-phoneme "word", audibly fragmenting all speech.
238 std::string joined;
239 for (const std::string& phoneme : phonemes) {
240 if (phoneme.empty()) continue;
241 if (!joined.empty()) joined += ' ';
242 joined += phoneme;
243 }
244 bool success = sayPhoneme(phonemeType, joined);
245
246 // Signal completion
247 if (completeCallback_) {
248 completeCallback_();
249 }
250
251 return success;
252 }
