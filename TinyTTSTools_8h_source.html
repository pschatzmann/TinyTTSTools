32// Desktop test-build convenience: example sketches are written against a

34// hardware) we transparently alias I2SStream to AudioTools' desktop
35// MiniAudioStream backend instead -- the sketch source needs no change.

131 std::vector<std::string> toPhonemes(const std::string& text) {
132 // Step 1: Text preprocessing
133 std::vector<std::string> words = Tokenizer::tokenize(text);
134 if (words.empty()) {
135 if (errorCallback_) {
136 errorCallback_("Empty text input");
137 }
138 return {};
139 }
140
141 // Step 2: Grapheme-to-phoneme conversion
142 std::vector<std::string> phonemes;
143 for (const std::string& word : words) {
144 if (word == ".") {
145 phonemes.push_back("SIL"); // Add silence for punctuation
146 } else {
147 std::string wordPhonemes = g2p_->wordToPhonemes(word);
148 std::vector<std::string> wordPhonemeList =
149 StringUtils::split(wordPhonemes, ' ');
150 phonemes.insert(phonemes.end(), wordPhonemeList.begin(),
151 wordPhonemeList.end());
152 phonemes.push_back("SP"); // Add short pause between words
153 }
154 }
155
156 return phonemes;
157 }

210 bool sayPhonemes(PhonemeType phonemeType,
211 const std::vector<std::string>& phonemes) {
212 if (!initialized_) {
213 if (errorCallback_) {
214 errorCallback_("TTS not initialized");
215 }
216 return false;
217 }
218
219 if (phonemes.empty()) {
220 if (errorCallback_) {
221 errorCallback_("Empty phoneme sequence");
222 }
223 return false;
224 }
225
226 // Join into ONE space-separated sequence and make a single call, rather
227 // than calling sayPhoneme() once per individual phoneme token. Every
228 // vocoder that needs multi-phoneme context (DiphoneVocoder to form any
229 // diphone pair at all; PhonemeVocoder for cross-unit lookahead/
230 // coarticulation) only gets that context from the phonemes it sees
231 // *within one call* -- calling one phoneme at a time silently starved
232 // them of it: DiphoneVocoder in particular can't form a diphone at all
233 // from a single phoneme, so it fell back to treating every phoneme as
234 // its own isolated one-phoneme "word", audibly fragmenting all speech.
235 std::string joined;
236 for (const std::string& phoneme : phonemes) {
237 if (phoneme.empty()) continue;
238 if (!joined.empty()) joined += ' ';
239 joined += phoneme;
240 }
241 bool success = sayPhoneme(phonemeType, joined);
242
243 // Signal completion
244 if (completeCallback_) {
245 completeCallback_();
246 }
247
248 return success;
249 }
