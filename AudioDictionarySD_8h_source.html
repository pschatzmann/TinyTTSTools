71 if (file_size <= wav_header_size_) {
72 file.close();
73 return;
74 }
75
76 size_t audio_data_size = file_size - wav_header_size_;
77 size = audio_data_size;
78 size_t bytes_per_sample = bits / 8;
79 length_ = audio_data_size / bytes_per_sample;
80
81 if (length_ == 0) {
82 file.close();
83 return;
84 }
85
86 // Skip WAV header
87 file.seek(wav_header_size_);
88
89 // Calculate raw data size in bytes
90 size_t raw_data_size = length_ * bytes_per_sample;
91
92 // Resize vector to hold raw audio data
93 raw_data_.resize(raw_data_size);
94
95 // Read raw audio data as bytes
96 size_t bytes_read = file.read(raw_data_.data(), raw_data_size);
97
98 // Resize to actual bytes read if incomplete
99 if (bytes_read < raw_data_size) {
100 raw_data_.resize(bytes_read);
101 // Update length_ based on actual data read
102 length_ = bytes_read / bytes_per_sample;
103 }
104
105 file.close();

108 data = raw_data_.data();
