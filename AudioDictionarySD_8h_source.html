65 mutable std::vector<uint8_t, Allocator>
66 raw_data_; // Vector to store raw audio data as bytes
67
68 void loadAudioData() {
69 // Reset previous data
70 length_ = 0;
71 size = 0;
72
73 File file = SD.open(file_path_, FILE_READ);
74 if (!file) {
75 TTS_LOGE("SDSoundEntry: failed to open '%s'", file_path_);
76 return;
77 }
78
79 // Calculate length from file size
80 size_t file_size = file.size();
81 if (file_size <= wav_header_size_) {
82 TTS_LOGE("SDSoundEntry: '%s' is %zu bytes, too small for a %zu-byte "
83 "header", file_path_, file_size, wav_header_size_);
84 file.close();
85 return;
86 }
87
88 size_t audio_data_size = file_size - wav_header_size_;
89 size = audio_data_size;
90 size_t bytes_per_sample = bits / 8;
91 if (bytes_per_sample == 0) {
92 // bits < 8 (e.g. 4 for IMA-ADPCM) isn't a raw-PCM format this class
93 // supports (see class doc) -- would otherwise divide by zero below.
94 TTS_LOGE("SDSoundEntry: unsupported bits=%u for '%s' (this class only "
95 "reads raw PCM8/PCM16, not compressed formats)",
96 static_cast<unsigned>(bits), file_path_);
97 file.close();
98 return;
99 }
100 length_ = audio_data_size / bytes_per_sample;
101
102 if (length_ == 0) {
103 TTS_LOGE("SDSoundEntry: '%s' has no audio data after the header",
104 file_path_);
105 file.close();
106 return;
107 }
108
109 // Skip WAV header
110 file.seek(wav_header_size_);
111
112 // Calculate raw data size in bytes
113 size_t raw_data_size = length_ * bytes_per_sample;
114
115 // Resize vector to hold raw audio data
116 raw_data_.resize(raw_data_size);
117
118 // Read raw audio data as bytes
119 size_t bytes_read = file.read(raw_data_.data(), raw_data_size);
120
121 // Resize to actual bytes read if incomplete
122 if (bytes_read < raw_data_size) {
123 TTS_LOGW("SDSoundEntry: '%s' short read (%zu of %zu bytes)",
124 file_path_, bytes_read, raw_data_size);
125 raw_data_.resize(bytes_read);
126 // Update length_ based on actual data read
127 length_ = bytes_read / bytes_per_sample;
128 }
129
130 file.close();
131
132 // Set data pointer for parent class access
133 data = raw_data_.data();
134 }
135};

192 SoundEntry* getSoundEntry(const char* phoneme) override {

196 file_path_.clear();
