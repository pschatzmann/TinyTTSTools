43 static std::string preprocessText(const std::string& text) {
44 std::string result = StringUtils::toLowerCase(text);
45
46 // Remove punctuation and normalize
47 std::string normalized;
48 for (unsigned char c : result) {
49 if (c >= 0x80) {
50 // Continuation/lead byte of a multi-byte UTF-8 sequence -- never
51 // a valid ASCII letter/space/punctuation on its own, so <cctype>
52 // (ASCII-only) can't classify it. Passing it to isalpha()/
53 // isspace() as a plain (possibly signed) char is also undefined
54 // behavior per the standard (they only accept values representable
55 // as unsigned char or EOF). Pass it through untouched instead of
56 // running it through <cctype> -- dropping it would corrupt any
57 // accented word (see PhonemeDictionaryFR/DE/ES.h, whose entries
58 // are keyed by their UTF-8-accented spelling, e.g. "café", "chéri").
59 normalized += static_cast<char>(c);
60 } else if (std::isalpha(c) || std::isspace(c)) {
61 normalized += static_cast<char>(c);
62 } else if (c == '.' || c == '!' || c == '?') {
63 normalized += " . "; // Convert sentence endings to pauses
64 }
65 }
66
67 return StringUtils::trim(normalized);
68 }
