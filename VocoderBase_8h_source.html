149 void outputSilence(int sampleRateHz, uint16_t durationMs, ::Print& out) {
150 if (durationMs == 0) return;
151
152 // Calculate number of samples needed for the duration
153 size_t numSamples = (static_cast<size_t>(durationMs) * sampleRateHz) / 1000;
154
155 // Output zero values (16-bit signed PCM silence)
156 int16_t silence = 0;
157 const uint8_t* silenceBytes = reinterpret_cast<const uint8_t*>(&silence);
158
159 for (size_t i = 0; i < numSamples; i++) {
160 out.write(silenceBytes, sizeof(int16_t));
161 }

186 bool processPhonemeSequence(const std::string& phoneme, ::Print& out,
187 const PhonemeSynthesisParams& params) {
188 if (!isReady()) {
189 return false;
190 }
191
192 // If input contains multiple phonemes (space-separated), process them as a sequence
193 std::vector<std::string> phonemes = StringUtils::split(phoneme, ' ');
194
195 if (phonemes.empty()) {
196 return false;
197 }
198
199 // Process each phoneme individually
200 bool success = true;
201 for (const std::string& singlePhoneme : phonemes) {
202 if (!processSinglePhoneme(singlePhoneme, out, params)) {
203 success = false;

273 uint16_t getPhonemeWithStressDuration(const std::string& phoneme, int stress = 0) {
274 uint16_t baseDuration = translator_.getPhonemeDuration(defaultPhonemeType, phoneme);
275
276 // Apply stress duration modification (stressed syllables are typically longer)
277 if (stress == 1) {
278 return static_cast<uint16_t>(baseDuration * 1.2f); // 20% longer for primary stress
279 } else if (stress == 2) {
