257 void outputSilence(int sampleRateHz, uint16_t durationMs, ::Print& out) {
258 if (durationMs == 0) return;
259
260 // Calculate number of samples needed for the duration
261 size_t numSamples = (static_cast<size_t>(durationMs) * sampleRateHz) / 1000;
262
263 // Output zero values (16-bit signed PCM silence)
264 int16_t silence = 0;
265 const uint8_t* silenceBytes = reinterpret_cast<const uint8_t*>(&silence);

267 for (size_t i = 0; i < numSamples; i++) {
268 out.write(silenceBytes, sizeof(int16_t));

349 static uint16_t resolveDuration(uint16_t defaultDurationMs, const PhonemeSynthesisParams& params) {
350 if (params.durationMs > 0) return params.durationMs;
351 if (params.speed > 0.0f && params.speed != 1.0f) {
352 float scaled = defaultDurationMs / params.speed;
353 if (scaled < 1.0f) scaled = 1.0f;
354 return static_cast<uint16_t>(scaled + 0.5f);
355 }
356 return defaultDurationMs;
357 }
