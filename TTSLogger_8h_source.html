86 void log(TTSLogLevel level, const char* format, ...) {
87 if (level < logLevel_) {
88 return; // Message level is below threshold
89 }
90
91 if (enabled_ && format) {
92 va_list args;
93 va_start(args, format);
94
95 // Format the message into a buffer
96 char buffer[TTS_LOG_BUFFER_SIZE];
97 vsnprintf(buffer, sizeof(buffer), format, args);
98 buffer[sizeof(buffer) - 1] = '\0'; // Ensure null termination
99
100 va_end(args);
101
102 outputMessage(level, buffer);
103 }
104 }

149// A C++17 inline variable, not `static` -- `static` at header scope gives
150// every including translation unit its own separate instance (internal
151// linkage), so a `TTSLogger.begin(...)` call in one .cpp would silently
152// have no effect on the instance any other .cpp/.ino sees. `inline` gives
153// the whole program exactly one shared instance, which is what a global
154// logger needs to actually be global.
155inline TTSLoggerClass TTSLogger;
