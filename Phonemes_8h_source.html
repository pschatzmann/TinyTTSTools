249 const PhonemeInfo* getPhoneme(PhonemeType phonemeType,
250 const std::string& phoneme) const {
251 for (const auto& entry : phoneme_map) {
252 const char* targetPhoneme = nullptr;
253
254 // Get the phoneme representation from the mapping entry
255 switch (phonemeType) {
256 case PhonemeType::ARPAbet:
257 targetPhoneme = entry.arpabet;
258 break;
259 case PhonemeType::IPA:
260 targetPhoneme = entry.ipa;
261 break;
262 case PhonemeType::XSAMPA:
263 targetPhoneme = entry.xsampa;
264 break;
265 }
266
267 // Check if we found a match
268 if (targetPhoneme && targetPhoneme == phoneme) {
269 return &entry;
270 }
271 }
272 return nullptr; // Phoneme not found
273 }

301 const char* translatePhoneme(PhonemeType sourceType,
302 const std::string& phoneme,
303 PhonemeType targetType) const {
304 // Use getPhoneme to find the entry
305 const PhonemeInfo* entry = getPhoneme(sourceType, phoneme);
306 if (!entry) {
307 return nullptr; // Phoneme not found in mapping table
308 }
309
310 // Return the target representation
311 switch (targetType) {
312 case PhonemeType::ARPAbet:
313 return entry->arpabet;
314 case PhonemeType::IPA:
315 return entry->ipa;
316 case PhonemeType::XSAMPA:
317 return entry->xsampa;
318 }
319 return nullptr;
320 }

358 std::string translatePhonemes(PhonemeType sourceType,
359 const std::string& phonemes,
360 PhonemeType targetType) const {
361 std::string result;
362 std::string currentPhoneme;
363 bool firstPhoneme = true;
364
365 // Parse the input string character by character
366 for (size_t i = 0; i <= phonemes.length(); ++i) {
367 // Check if we've reached a space or the end of string
368 if (i == phonemes.length() || phonemes[i] == ' ') {
369 if (!currentPhoneme.empty()) {
370 // Translate the current phoneme
371 const char* translatedPhoneme =
372 translatePhoneme(sourceType, currentPhoneme, targetType);
373 if (translatedPhoneme) {
374 // Add space separator between phonemes (except for the first one)
375 if (!firstPhoneme) {
376 result += " ";
377 }
378 result += translatedPhoneme;
379 firstPhoneme = false;
380 }
381 // Reset for next phoneme
382 currentPhoneme.clear();
383 }
384 } else {
385 // Build up the current phoneme
386 currentPhoneme += phonemes[i];
387 }
388 }
389
390 return result;
391 }
