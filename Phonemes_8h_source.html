129 const PhonemeInfo* getPhoneme(PhonemeType phonemeType,
130 const std::string& phoneme) const {
131 for (const auto& entry : phoneme_map) {
132 const char* targetPhoneme = nullptr;
133
134 // Get the phoneme representation from the mapping entry
135 switch (phonemeType) {
136 case PhonemeType::ARPAbet:
137 targetPhoneme = entry.arpabet;
138 break;
139 case PhonemeType::IPA:
140 targetPhoneme = entry.ipa;
141 break;
142 case PhonemeType::XSAMPA:
143 targetPhoneme = entry.xsampa;
144 break;
145 }
146
147 // Check if we found a match
148 if (targetPhoneme && targetPhoneme == phoneme) {
149 return &entry;
150 }
151 }
152 return nullptr; // Phoneme not found
153 }

238 std::string translatePhonemes(PhonemeType sourceType,
239 const std::string& phonemes,
240 PhonemeType targetType) const {
241 std::string result;
242 std::string currentPhoneme;
243 bool firstPhoneme = true;
244
245 // Parse the input string character by character
246 for (size_t i = 0; i <= phonemes.length(); ++i) {
247 // Check if we've reached a space or the end of string
248 if (i == phonemes.length() || phonemes[i] == ' ') {
249 if (!currentPhoneme.empty()) {
250 // Translate the current phoneme
251 const char* translatedPhoneme =
252 translatePhoneme(sourceType, currentPhoneme, targetType);
253 if (translatedPhoneme) {
254 // Add space separator between phonemes (except for the first one)
255 if (!firstPhoneme) {
256 result += " ";
257 }
258 result += translatedPhoneme;
259 firstPhoneme = false;
260 }
261 // Reset for next phoneme
262 currentPhoneme.clear();
263 }
264 } else {
265 // Build up the current phoneme
266 currentPhoneme += phonemes[i];
267 }
268 }
269
270 return result;
271 }

356 std::string toArpabetString(Phone phone) const {
357 uint8_t id = static_cast<uint8_t>(phone);
358 if (id < 43) {
359 const PhonemeInfo* info = getPhonemeById(id);
360 return info ? std::string(info->arpabet) : std::string();
361 }
362 if (id > 72) return std::string();
363 // Stress-variant ids: 15 vowels x 2 (primary/secondary), in the same
364 // order as declared in Phone (43=AA1, 44=AA2, 45=AE1, ..., 72=UW2).
365 static const char* const kStressVowels[15] = {
366 "AA", "AE", "AH", "AO", "AW", "AY", "EH", "ER",
367 "EY", "IH", "IY", "OW", "OY", "UH", "UW"};
368 size_t offset = id - 43;
369 size_t vowelIndex = offset / 2;
370 int stress = static_cast<int>(offset % 2) + 1; // 1 or 2
371 return std::string(kStressVowels[vowelIndex]) + static_cast<char>('0' + stress);
372 }
