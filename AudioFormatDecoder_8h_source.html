89 static constexpr size_t kSamplesPerBlock = (kBlockAlign - 4) * 2 + 1;

91 size_t sampleCount(const uint8_t* data, size_t byteSize) const override {
92 (void)data;
93 size_t fullBlocks = byteSize / kBlockAlign;
94 size_t remBytes = byteSize % kBlockAlign;
95 size_t count = fullBlocks * kSamplesPerBlock;
96 if (remBytes >= 4) {
97 count += (remBytes - 4) * 2 + 1;
98 }
99 return count;
100 }

102 int16_t sampleAt(const uint8_t* data, size_t byteSize, size_t index) const override {
103 (void)byteSize;
104 size_t block = index / kSamplesPerBlock;
105 size_t withinBlock = index % kSamplesPerBlock;
106 const uint8_t* blockData = data + block * kBlockAlign;
107
108 int32_t predictor = static_cast<int16_t>(
109 static_cast<uint16_t>(blockData[0]) | (static_cast<uint16_t>(blockData[1]) << 8));
110 if (withinBlock == 0) return static_cast<int16_t>(predictor);
111
112 int8_t stepIndex = static_cast<int8_t>(blockData[2]);
113 for (size_t i = 1; i <= withinBlock; ++i) {
114 size_t nibbleIndex = i - 1;
115 uint8_t byte = blockData[4 + nibbleIndex / 2];
116 uint8_t code = (nibbleIndex % 2 == 0) ? (byte & 0x0F) : (byte >> 4);
117 predictor = decodeStep(code, predictor, stepIndex);
118 }
119 return static_cast<int16_t>(predictor);
120 }
121
122 private:
123 static int32_t decodeStep(uint8_t code, int32_t predictor, int8_t& stepIndex) {
124 static constexpr int16_t kStepTable[89] = {
125 7, 8, 9, 10, 11, 12, 13, 14, 16, 17, 19,
126 21, 23, 25, 28, 31, 34, 37, 41, 45, 50, 55,
127 60, 66, 73, 80, 88, 97, 107, 118, 130, 143, 157,
128 173, 190, 209, 230, 253, 279, 307, 337, 371, 408, 449,
129 494, 544, 598, 658, 724, 796, 876, 963, 1060, 1166, 1282,
130 1411, 1552, 1707, 1878, 2066, 2272, 2499, 2749, 3024, 3327, 3660,
131 4026, 4428, 4871, 5358, 5894, 6484, 7132, 7845, 8630, 9493, 10442,
132 11487, 12635, 13899, 15289, 16818, 18500, 20350, 22385, 24623, 27086, 29794,
133 32767};
134 static constexpr int8_t kIndexTable[16] = {-1, -1, -1, -1, 2, 4, 6, 8,
135 -1, -1, -1, -1, 2, 4, 6, 8};
136
137 int32_t step = kStepTable[stepIndex];
138 int32_t diff = step >> 3;
139 if (code & 4) diff += step;
140 if (code & 2) diff += step >> 1;
141 if (code & 1) diff += step >> 2;
142 if (code & 8) predictor -= diff;
143 else predictor += diff;
144
145 if (predictor > 32767) predictor = 32767;
146 if (predictor < -32768) predictor = -32768;
147
148 stepIndex = static_cast<int8_t>(stepIndex + kIndexTable[code]);
149 if (stepIndex < 0) stepIndex = 0;
150 if (stepIndex > 88) stepIndex = 88;
151
152 return predictor;
153 }
