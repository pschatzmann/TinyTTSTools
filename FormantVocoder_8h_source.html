284 void configureFormantFilter(FormantFilter& filter, float frequency,
285 float bandwidth) {
286 if (frequency <= 0.0f) { // handle silence
287 filter.a1 = 0.0f;
288 filter.a2 = 0.0f;
289 filter.b0 = 0.0f;
290 return;
291 }
292 float r = expf(-M_PI * bandwidth / sampleRate_); // pole radius
293 float theta = 2.0f * M_PI * frequency / sampleRate_;
294 // Canonical resonator: y[n] = b0*x[n] + a1*y[n-1] + a2*y[n-2]; poles at
295 // r*e^{±j theta}
296 filter.a1 = 2.0f * r * cosf(theta);
297 filter.a2 = -r * r;
298 filter.b0 =
299 1.0f -
300 r; // energy normalization keeping level stable across bandwidths
301 }

304 float applyFormantFilter(FormantFilter& filter, float input) {
305 float y = filter.b0 * input + filter.a1 * filter.y1 + filter.a2 * filter.y2;
306 filter.y2 = filter.y1;
307 filter.y1 = y;
308 return y;
309 }

312 float generateVoiceSource(float t, float f0Base) {
313 // Pitch with jitter updated every few milliseconds
314 const uint16_t jitterIntervalSamples = sampleRate_ / 200; // ~5ms
315 if (++jitterCounter_ >= jitterIntervalSamples) {
316 jitterCounter_ = 0;
317 float jitter = cfg_.jitterPct * frand();
318 // Base the jittered pitch on f0Base (the per-phoneme pitch passed in,
319 // which may carry a stress rise or an explicit pitchHz override from
320 // PhonemeSynthesisParams) rather than the instance's static voiceF0_ --
321 // using voiceF0_ here made f0Base a dead parameter and silently
322 // discarded any per-call pitch adjustment.
323 f0Current_ = f0Base * (1.0f + jitter);
324 }
325 // Add small downward pitch contour over phoneme duration (approx t
326 // fraction)
327 float dynamicF0 = f0Current_ - cfg_.pitchFallPerSec * t;
328 if (dynamicF0 < 50.0f) dynamicF0 = 50.0f;
329 voicePhase_ += 2.0f * M_PI * dynamicF0 / sampleRate_;
330 if (voicePhase_ >= 2.0f * M_PI) voicePhase_ -= 2.0f * M_PI;
331 float phase_norm = voicePhase_ / (2.0f * M_PI);
332 // Simplified LF derivative (two-piece polynomial) + closing spike
333 float glottal = 0.0f;
334 const float openFrac = 0.45f;
335 const float closeFrac = 0.8f;
336 if (phase_norm < openFrac) {
337 float x = phase_norm / openFrac;
338 glottal = x - x * x * x * 0.3333f; // gentle rise
339 } else if (phase_norm < closeFrac) {
340 float x = (phase_norm - openFrac) / (closeFrac - openFrac);
341 glottal = 1.0f - 0.5f * x * x; // quasi-plateau with slight fall
342 } else {
343 float x = (phase_norm - closeFrac) / (1.0f - closeFrac);
344 glottal = 0.2f * (1.0f - x); // abrupt closure & return to zero
345 }
346 // Shimmer (amplitude variation)
347 glottal *= (1.0f + cfg_.shimmerPct * frand());
348 // Spectral tilt
349 spectralTiltState_ = (1.0f - cfg_.spectralTilt) * glottal +
350 cfg_.spectralTilt * spectralTiltState_;
351 // Add controlled breathiness
352 float aspiration = cfg_.breathiness * generateNoiseSource();
353 return 0.55f * spectralTiltState_ + 0.35f * glottal + aspiration;
354 }

442 if (!(isNasal && cfg_.enableNasalNotch)) return;
443 if (i == 0) {
444 auto configNotch = [&](NotchFilter& nf, float f, float bw) {
445 float r = expf(-M_PI * bw / sampleRate_);
446 float theta = 2.0f * M_PI * f / sampleRate_;
447 nf.b0 = 1.0f;
448 nf.b1 = -2.0f * cosf(theta);
449 nf.b2 = 1.0f;
450 nf.a1 = 2.0f * r * cosf(theta);
451 nf.a2 = -r * r;
452 nf.x1 = nf.x2 = nf.y1 = nf.y2 = 0.0f;
453 nf.enabled = true;
454 };
455 configNotch(nasalNotch1_, cfg_.nasalNotchFreq1, cfg_.nasalNotchBw1);
456 configNotch(nasalNotch2_, cfg_.nasalNotchFreq2, cfg_.nasalNotchBw2);
457 }

459 if (!nf.enabled) return in;
460 float y = nf.b0 * in + nf.b1 * nf.x1 + nf.b2 * nf.x2 + nf.a1 * nf.y1 +
461 nf.a2 * nf.y2;
462 nf.x2 = nf.x1;
463 nf.x1 = in;
464 nf.y2 = nf.y1;
465 nf.y1 = y;
466 return y;
467 };
468 float before = sample;
469 sample = applyNotch(nasalNotch1_, sample);
470 sample = applyNotch(nasalNotch2_, sample);
471 sample =
472 (1.0f - cfg_.nasalNotchDepth) * before + cfg_.nasalNotchDepth * sample;

499 void synthesizeSamples(PhonemeSynthesisContext& ctx) {
500 size_t overlapSamples =
501 static_cast<size_t>((cfg_.overlapMs / 1000.0f) * sampleRate_);
502 if (overlapSamples > ctx.numSamples / 2)
503 overlapSamples = ctx.numSamples / 2;
504 for (size_t i = 0; i < ctx.numSamples; ++i) {
505 float t = (float)i / (float)sampleRate_;
506 float progress =
507 (ctx.numSamples > 1) ? (float)i / (float)(ctx.numSamples - 1) : 0.0f;
508 // Steady-state (or, for diphthongs, mid-phoneme-eased) formant
509 // target for this sample.
510 float tf1, tf2, tf3;
511 if (ctx.dynamic) {
512 float ease = progress < 0.5f
513 ? 2.0f * progress * progress
514 : -1.0f + (4.0f - 2.0f * progress) * progress;
515 tf1 = ctx.params.f1 + (ctx.target2.f1 - ctx.params.f1) * ease;
516 tf2 = ctx.params.f2 + (ctx.target2.f2 - ctx.params.f2) * ease;
517 tf3 = ctx.params.f3 + (ctx.target2.f3 - ctx.params.f3) * ease;
518 } else {
519 tf1 = ctx.params.f1;
520 tf2 = ctx.params.f2;
521 tf3 = ctx.params.f3;
522 }
523 // Coarticulation: for the first attackSamples, blend from where the
524 // PREVIOUS phoneme's formants ended (ctx.entryF1/2/3) toward this
525 // sample's own target, instead of starting already at rest -- real
526 // formant transitions between adjacent phonemes carry much of
527 // speech's intelligibility (especially consonant place-of-
528 // articulation cues), so snapping every phoneme instantly to its own
529 // target made speech sound like separately-glued blobs.
530 if (i < ctx.attackSamples) {
531 float attackProgress = (float)i / (float)ctx.attackSamples;
532 // Raised-cosine ease: smooth start and end of the glide.
533 float ease = 0.5f - 0.5f * cosf((float)M_PI * attackProgress);
534 tf1 = ctx.entryF1 + (tf1 - ctx.entryF1) * ease;
535 tf2 = ctx.entryF2 + (tf2 - ctx.entryF2) * ease;
536 tf3 = ctx.entryF3 + (tf3 - ctx.entryF3) * ease;
537 }
538 lastF1_ = formantSmoothingFactor_ * lastF1_ +
539 (1.0f - formantSmoothingFactor_) * tf1;
540 lastF2_ = formantSmoothingFactor_ * lastF2_ +
541 (1.0f - formantSmoothingFactor_) * tf2;
542 lastF3_ = formantSmoothingFactor_ * lastF3_ +
543 (1.0f - formantSmoothingFactor_) * tf3;
544 if ((i & 31) == 0) {
545 configureFormantFilter(f1Filter_, lastF1_, ctx.params.bw1);
546 configureFormantFilter(f2Filter_, lastF2_, ctx.params.bw2);
547 configureFormantFilter(f3Filter_, lastF3_, ctx.params.bw3);
548 }
549 float source = 0.0f;
550 if (ctx.flags.voiced) {
551 float f0Adj = ctx.f0;
552 // Pitch contour (see PhonemeSynthesisParams::f0StartRatio/
553 // MidRatio/EndRatio, settable directly or via a PhonemeModifier
554 // TONE_* value): piecewise-linear interpolation across the
555 // phoneme's own duration, all-1.0 (the default) leaving f0Adj
556 // unchanged.
557 float contourRatio =
558 progress < 0.5f
559 ? ctx.f0StartRatio +
560 (ctx.f0MidRatio - ctx.f0StartRatio) * (progress * 2.0f)
561 : ctx.f0MidRatio + (ctx.f0EndRatio - ctx.f0MidRatio) *
562 ((progress - 0.5f) * 2.0f);
563 f0Adj *= contourRatio;
564 if (ctx.stressPitchRise > 0.0f) {
565 float sd = progress / (cfg_.stressDecayPortion <= 0.01f
566 ? 0.01f
567 : cfg_.stressDecayPortion);
568 if (sd > 1.0f) sd = 1.0f;
569 float decay = expf(-2.2f * sd);
570 f0Adj += ctx.stressPitchRise * decay;
571 }
572 float g = generateVoiceSource(progress * ctx.duration, f0Adj);
573 float harmonics = 0.0f;
574 const int NH = 5;
575 for (int h = 2; h <= NH; ++h)
576 harmonics += (1.0f / (float)h) * sinf(voicePhase_ * h);
577 float voicedSource = 0.65f * g + 0.30f * harmonics;
578 if (ctx.voicing >= 1.0f) {
579 source = voicedSource;
580 } else {
581 // Blend toward an unvoiced/whispered version of this phoneme:
582 // replace the harmonic glottal source with broadband noise as
583 // voicing decreases from 1.0 (fully voiced) to 0.0 (whisper).
584 // Naturally unvoiced phonemes never reach this branch, so this
585 // only affects vowels/voiced consonants.
586 float n = generateNoiseSource();
587 lp_hp_ = 0.85f * lp_hp_ + 0.15f * n;
588 float hp = n - lp_hp_;
589 float clampedVoicing = ctx.voicing < 0.0f ? 0.0f : ctx.voicing;
590 source = voicedSource * clampedVoicing + hp * (1.0f - clampedVoicing);
591 }
592 } else {
593 float n = generateNoiseSource();
594 lp_hp_ = 0.85f * lp_hp_ + 0.15f * n;
595 float hp = n - lp_hp_;
596 if (ctx.flags.strongSibilant) {
597 float bp = hp - 0.6f * prevNoise1_;
598 prevNoise1_ = hp;
599 source = cfg_.sibilantGain * bp;
600 } else if (ctx.flags.postalveolar) {
601 float bp = (hp - prevNoise2_);
602 prevNoise2_ = hp;
603 source = cfg_.postalveolarGain * bp;
604 } else
605 source = hp;
606 }
607 float out1 = applyFormantFilter(f1Filter_, source) * ctx.params.a1;
608 float out2 = applyFormantFilter(f2Filter_, source) * ctx.params.a2;
609 float out3 = applyFormantFilter(f3Filter_, source) * ctx.params.a3;
610 float out4 = 0.0f;
611 if (cfg_.enableF4) out4 = applyFormantFilter(f4Filter_, source) * 0.25f;
612 if (!ctx.flags.voiced && ctx.flags.strongSibilant)
613 out3 *= cfg_.sibilantHFFormantScale;
614 float sample = out1 + out2 + out3 + out4;
615 applyNasalNotchesIfNeeded(sample, ctx.flags.nasal, i);
616 sample = 0.9f * sample + 0.1f * prevSample_;
617 prevSample_ = sample;
618 float envelope = generateEnvelope(t, ctx.duration);
619 sample *= envelope;
620 finalizeSampleNasalLP(sample, ctx.flags.nasal);
621 applyDeEsserIfNeeded(sample, ctx.flags);
622 if (sample > 1.0f) sample = 1.0f - 0.5f * (sample - 1.0f);
623 if (sample < -1.0f) sample = -1.0f - 0.5f * (sample + 1.0f);
624 audioBuffer_[i] = (int16_t)(sample * 15000.0f);
625 }
626 }

629 void applyRMSNormalization(size_t numSamples, const PhonemeFlags& flags) {
630 int64_t sumSq = 0;
631 for (size_t i = 0; i < numSamples; ++i) {
632 int32_t v = audioBuffer_[i];
633 sumSq += (int64_t)v * v;
634 }
635 float rms =
636 (numSamples > 0) ? sqrtf((float)sumSq / (float)numSamples) : 0.0f;
637 float target = flags.voiced ? cfg_.targetRMS
638 : ((flags.strongSibilant || flags.postalveolar)
639 ? cfg_.fricativeTargetRMS
640 : cfg_.targetRMS * 0.9f);
641 if (rms > 1.0f) {
642 float desiredGain = target / rms;
643 if (desiredGain > 1.5f) desiredGain = 1.5f;
644 adaptiveGain_ = (1.0f - cfg_.rmsAdaptRate) * adaptiveGain_ +
645 cfg_.rmsAdaptRate * desiredGain;
646 for (size_t i = 0; i < numSamples; ++i) {
647 int32_t v = (int32_t)(audioBuffer_[i] * adaptiveGain_);
648 if (v > 32767)
649 v = 32767;

658 void applySibilantPostProcessing(size_t numSamples,
659 const PhonemeFlags& flags) {
660 if (flags.voiced || !(flags.strongSibilant || flags.postalveolar)) return;
661 // Peak limiting
662 int16_t peak = 0;
663 for (size_t i = 0; i < numSamples; ++i) {
664 int16_t v = audioBuffer_[i];
665 int16_t a = v < 0 ? -v : v;
666 if (a > peak) peak = a;
667 }
668 float targetP = cfg_.fricativeMaxPeak;
669 if (peak > targetP && peak > 0) {
670 float scale = targetP / (float)peak;
671 for (size_t i = 0; i < numSamples; ++i) {
672 audioBuffer_[i] = (int16_t)(audioBuffer_[i] * scale);
673 }
674 }
675 // Dynamic high-band compression
676 float lp = 0.0f;
677 float sumSqHB = 0.0f;
678 for (size_t i = 0; i < numSamples; ++i) {
679 float x = (float)audioBuffer_[i];
680 lp = lp + 0.08f * (x - lp);
681 float hb = x - lp;
682 sumSqHB += hb * hb;
683 }
684 float rmsHB =
685 (numSamples > 0) ? sqrtf(sumSqHB / numSamples) : 0.0f;
686 float targetHB = cfg_.sibilantDynamicTargetRMS;
687 if (rmsHB > targetHB && rmsHB > 1.0f) {
688 float over = rmsHB / targetHB;
689 float expn = (1.0f / cfg_.sibilantCompressionRatio) - 1.0f;
690 float desiredGain = powf(over, expn);
691 if (desiredGain < 0.25f) desiredGain = 0.25f;
692 float smooth = (desiredGain < sibilantDynGain_) ? cfg_.sibilantAttack
693 : cfg_.sibilantRelease;
694 sibilantDynGain_ =
695 smooth * sibilantDynGain_ + (1.0f - smooth) * desiredGain;
696 for (size_t i = 0; i < numSamples; ++i) {
697 int32_t v = (int32_t)(audioBuffer_[i] * sibilantDynGain_);
698 if (v > 32767)
699 v = 32767;
700 else if (v < -32768)
701 v = -32768;
702 audioBuffer_[i] = (int16_t)v;
703 }
704 } else {
705 sibilantDynGain_ = cfg_.sibilantRelease * sibilantDynGain_ +
706 (1.0f - cfg_.sibilantRelease);
707 }
708 if (cfg_.sibilantLowpassMix > 0.0f) {
709 float lp2 = 0.0f;
710 float mix = cfg_.sibilantLowpassMix;
711 for (size_t i = 0; i < numSamples; ++i) {
712 float x = (float)audioBuffer_[i];
713 lp2 += 0.12f * (x - lp2);
714 float softened = lp2;
715 float y = x * (1.0f - mix) + softened * mix;
716 int32_t v = (int32_t)(y);
717 if (v > 32767)
718 v = 32767;
719 else if (v < -32768)
720 v = -32768;
721 audioBuffer_[i] = (int16_t)v;
722 }
723 }
724 if (cfg_.sibilantFinalGain < 0.999f) {
725 float g = cfg_.sibilantFinalGain;
726 for (size_t i = 0; i < numSamples; ++i) {
727 int32_t v = (int32_t)(audioBuffer_[i] * g);
728 if (v > 32767)
729 v = 32767;
730 else if (v < -32768)
731 v = -32768;
732 audioBuffer_[i] = (int16_t)v;
733 }
734 }

752 void applyCrossfade(size_t numSamples) {
753 if (!previousTail_.empty()) {
754 size_t overlap = std::min(previousTail_.size(), numSamples);
755 for (size_t i = 0; i < overlap; ++i) {
756 float w = (float)i / (float)(overlap - 1);
757 float a = previousTail_[i] * (1.0f - w);
758 float b = audioBuffer_[i] * w;
759 int32_t mix = (int32_t)a + (int32_t)b;
760 if (mix > 32767) mix = 32767;
761 if (mix < -32768) mix = -32768;
762 audioBuffer_[i] = (int16_t)mix;
763 }
764 previousTail_.clear();
765 }
766 size_t overlapSamples =
767 static_cast<size_t>((cfg_.overlapMs / 1000.0f) * sampleRate_);
768 if (overlapSamples > 8 && overlapSamples < numSamples) {
769 // Bound the capture to the samples actually written for THIS phoneme
770 // (numSamples), not audioBuffer_.end() -- audioBuffer_ never shrinks,
771 // so its size() can exceed numSamples whenever an earlier phoneme was
772 // longer, and audioBuffer_.end() would then point past what was just
773 // written, capturing stale leftover data as the crossfade tail.
774 previousTail_.assign(audioBuffer_.begin() + (numSamples - overlapSamples),
775 audioBuffer_.begin() + numSamples);
776 }

809 void preparePhonemeSynthesis(const std::string& phoneme,
810 PhonemeSynthesisContext& ctx,
811 const PhonemeSynthesisParams& params) {
812 ctx.phonStr = stripStress(phoneme, ctx.stress);
813
814 // Modifier tags (see PhonemeModifiers.h): a token's own "T_j"-style tag
815 // takes priority over whatever the caller set on params.modifier for
816 // the whole call. A stress tag folds into ctx.stress (the existing
817 // mechanism -- computeStressPitchRise()/applyStressEnergyBoost()
818 // already handle it more precisely than deriveModifierEffect() would)
819 // instead of staying in ctx.modifier, so it's never double-applied.
820 PhonemeModifier tokenMod = PhonemeModifier::MOD_NONE;
821 ctx.phonStr = parsePhonemeModifiers(ctx.phonStr, tokenMod);
822 if (tokenMod == PhonemeModifier::MOD_STRESS_PRIMARY) {
823 ctx.stress = 1;
824 tokenMod = PhonemeModifier::MOD_NONE;
825 } else if (tokenMod == PhonemeModifier::MOD_STRESS_SECONDARY) {
826 ctx.stress = 2;
827 tokenMod = PhonemeModifier::MOD_NONE;
828 }
829 ctx.modifier =
830 tokenMod != PhonemeModifier::MOD_NONE ? tokenMod : params.modifier;
831
832 uint16_t defaultDurationMs =
833 static_cast<uint16_t>(getPhoneDuration(ctx.phonStr) * 1000.0f + 0.5f);
834 // MOD_ASPIRATED/MOD_SYLLABIC extend the phoneme's own natural duration
835 // directly (a VOT gap before the next phoneme's voicing onset, or
836 // syllable-nucleus time absorbed into a syllabic consonant) --
837 // additive, not a speed multiplier, since both are a fixed extra
838 // chunk of time rather than a proportional stretch.
839 if (ctx.modifier == PhonemeModifier::MOD_ASPIRATED) {
840 defaultDurationMs += 45;
841 } else if (ctx.modifier == PhonemeModifier::MOD_SYLLABIC) {
842 defaultDurationMs += 40;
843 }
844 // cfg_.speedScale is this voice's own default rate; a caller's own
845 // params.speed still composes with (multiplies) it rather than being
846 // overridden by it. MOD_LONG/MOD_HALF_LONG apply the same way, via
847 // deriveModifierEffect()'s speedMul (see PhonemeModifiers.h) -- shared
848 // with PhonemeSynthesisParams::setPhonemeModifier() so a caller-level
849 // and a token-level modifier stretch a phoneme by the same amount.
850 ModifierEffect modEffect = deriveModifierEffect(ctx.modifier);
851 PhonemeSynthesisParams effectiveParams = params;
852 effectiveParams.speed *= cfg_.speedScale * modEffect.speedMul;
853 ctx.duration = resolveDuration(defaultDurationMs, effectiveParams) / 1000.0f;
854 ctx.numSamples = (size_t)(ctx.duration * sampleRate_);
855 if (ctx.numSamples > audioBuffer_.size())
856 audioBuffer_.resize(ctx.numSamples);
857 ctx.params = getFormantParams(ctx.phonStr);
858 // SAM-style "mouth"/"throat" scaling: a uniform F1/F2 multiplier across
859 // every phoneme, layered on top of FormantRules.h's own per-phoneme
860 // values (see FormantVoiceConfig::mouthScale/throatScale doc).
861 ctx.params.f1 *= cfg_.mouthScale;
862 ctx.params.f2 *= cfg_.throatScale;
863 // Secondary-articulation/place-shifting modifiers (see
864 // applyModifierFormantShift()'s own doc for the per-modifier amounts)
865 // -- applied before the diphthong target copy below so a shifted
866 // vowel's diphthong glide target shifts consistently with it.
867 applyModifierFormantShift(ctx.params, ctx.modifier);
868 ctx.target2 = ctx.params;
869 ctx.dynamic = false;
870 setupDiphthongTargets(ctx.phonStr, ctx.target2, ctx.dynamic);
871 // Diphthong targets are looked up as their own independent
872 // FormantParams (not derived from ctx.params), so they need the same
873 // mouth/throat scaling applied separately.
874 if (ctx.dynamic) {
875 ctx.target2.f1 *= cfg_.mouthScale;
876 ctx.target2.f2 *= cfg_.throatScale;
877 }
878 randomizeFormantsIfNeeded(ctx.params, ctx.phonStr);
879 configureInitialFilters(ctx.params);
880 // Snapshot where the previous phoneme's formants ended up (see the
881 // coarticulation comment on PhonemeSynthesisContext), and size an
882 // attack window to glide from there to this phoneme's own target: up
883 // to ~30ms, capped at 30% of this phoneme's own duration so very short
884 // stops don't spend their whole duration transitioning.
885 ctx.entryF1 = lastF1_;
886 ctx.entryF2 = lastF2_;
887 ctx.entryF3 = lastF3_;
888 float attackSec = 0.03f < ctx.duration * 0.3f ? 0.03f : ctx.duration * 0.3f;
889 ctx.attackSamples = (size_t)(attackSec * sampleRate_);
890 if (ctx.attackSamples < 1) ctx.attackSamples = 1;
891 ctx.flags = classifyPhoneme(ctx.phonStr);
892 // MOD_NASALIZED/MOD_DEVOICED/MOD_VOICED override the base rule's own
893 // classification -- e.g. a normally-oral vowel tagged "~" gets nasal
894 // coupling, a normally-voiceless stop tagged "_v" gets sourced from
895 // the voiced glottal pulse instead of noise.
896 if (ctx.modifier == PhonemeModifier::MOD_NASALIZED) {
897 ctx.flags.nasal = true;
898 ctx.params.a1 *= 0.85f; // damp F1 amplitude, matches nasal vowels'
899 // own FormantRules.h entries
900 } else if (ctx.modifier == PhonemeModifier::MOD_DEVOICED) {
901 ctx.flags.voiced = false;
902 } else if (ctx.modifier == PhonemeModifier::MOD_VOICED) {
903 ctx.flags.voiced = true;
904 }
905 ctx.f0 = params.pitchHz > 0.0f ? params.pitchHz : voiceF0_;
906 if (ctx.params.f1 < 350.0f)
907 ctx.f0 *= 1.05f; // slight boost for closed vowels
908 if (ctx.modifier == PhonemeModifier::MOD_CREAKY) {
909 // Creaky voice's real signature is glottal-pulse irregularity, which
910 // this vocoder has no per-phoneme source-model hook for (see
911 // PhonemeModifierBits' own @note) -- lowering F0 is the one part of
912 // the effect a simple multiply can approximate honestly.
913 ctx.f0 *= 0.75f;
914 }
915 ctx.stressPitchRise = computeStressPitchRise(ctx.stress);
916 ctx.voicing = params.voicing * modEffect.voicingMul;
917 if (modEffect.voicingIsAbsolute) ctx.voicing = modEffect.voicingAbsolute;
918 // Compose the caller's own contour (params, default flat 1.0/1.0/1.0)
919 // with a TONE_* modifier's contour (modEffect, likewise default flat)
920 // multiplicatively -- when only one side is non-flat (the normal
921 // case), this reduces to using that side's ratios untouched.
922 ctx.f0StartRatio = params.f0StartRatio * modEffect.f0StartRatio;
923 ctx.f0MidRatio = params.f0MidRatio * modEffect.f0MidRatio;
924 ctx.f0EndRatio = params.f0EndRatio * modEffect.f0EndRatio;
925 }

966 bool handleSilenceSetup(PhonemeSynthesisContext& ctx, ::Print& out) {
967 if (!ctx.flags.silence) return false;
968
969 size_t numSamples = ctx.numSamples;
970 if (numSamples == 0) return true;
971
972 // Determine overlap samples (mirror logic from applyCrossfade)
973 size_t overlapSamples = static_cast<size_t>((cfg_.overlapMs / 1000.0f) * sampleRate_);
974 if (overlapSamples > 8 && overlapSamples < numSamples) {
975 // we'll set previousTail_ at end to zeros of this size
976 } else {
977 overlapSamples = 0; // treat as no tail storage
978 }
979
980 // 1. If there is a previous tail, perform crossfade (fade it out to zero)
981 if (!previousTail_.empty()) {
982 size_t overlap = std::min(previousTail_.size(), numSamples);
983 if (overlap > 0) {
984 // Create a small temporary buffer just for the overlap mix
985 std::vector<int16_t> temp(overlap);
986 for (size_t i = 0; i < overlap; ++i) {
987 float w = (float)i / (float)(overlap - 1);
988 // previous contribution fades out; silence fades in (0)
989 float a = previousTail_[i] * (1.0f - w);
990 int32_t v = (int32_t)a;
991 if (v > 32767) v = 32767; else if (v < -32768) v = -32768;
992 temp[i] = (int16_t)v;
993 }
994 out.write(reinterpret_cast<const uint8_t*>(temp.data()), overlap * sizeof(int16_t));
995 numSamples -= overlap;
996 }
997 previousTail_.clear();
998 }
999
1000 // 2. Stream the remaining silence as zero chunks without allocating a giant buffer
1001 static int16_t zeroChunk[256] = {0};
1002 while (numSamples > 0) {
1003 size_t n = (numSamples > 256) ? 256 : numSamples;
1004 out.write(reinterpret_cast<const uint8_t*>(zeroChunk), n * sizeof(int16_t));
1005 numSamples -= n;
1006 }
1007
1008 // 3. Prepare a zero tail for next phoneme if overlap policy applies
1009 if (overlapSamples > 0) {
1010 previousTail_.assign(overlapSamples, 0);
1011 }
1012 return true;
1013 }

1036 float generateEnvelope(float t, float duration) {
1037 float progress = t / duration;
1038
1039 // Different envelopes for different phoneme types
1040 if (duration < 0.08f) {
1041 // Short phonemes (consonants) - quick attack/decay
1042 if (progress < 0.3f) {
1043 return progress / 0.3f; // Quick attack
1044 } else if (progress > 0.7f) {
1045 return (1.0f - progress) / 0.3f; // Quick decay
1046 } else {
1047 return 1.0f; // Brief sustain
1048 }
1049 } else {
1050 // Longer phonemes (vowels) - more gradual envelope
1051 float attack = 0.15f;
1052 float release = 0.25f;
1053
1054 if (progress < attack) {
1055 // Smooth attack with exponential curve
1056 float attack_progress = progress / attack;
1057 return 1.0f - expf(-5.0f * attack_progress);
1058 } else if (progress > (1.0f - release)) {
1059 // Smooth release with exponential decay
1060 float release_progress = (progress - (1.0f - release)) / release;
1061 return expf(-3.0f * release_progress);
1062 } else {
1063 // Sustain with slight amplitude variation
1064 float sustain_variation = 1.0f + 0.05f * sinf(2.0f * (float)M_PI * t * 4.0f);
1065 return 0.95f * sustain_variation;
1066 }
