247 void configureFormantFilter(FormantFilter& filter, float frequency,
248 float bandwidth) {
249 if (frequency <= 0.0f) { // handle silence
250 filter.a1 = 0.0f;
251 filter.a2 = 0.0f;
252 filter.b0 = 0.0f;
253 return;
254 }
255 float r = expf(-M_PI * bandwidth / sampleRate_); // pole radius
256 float theta = 2.0f * M_PI * frequency / sampleRate_;
257 // Canonical resonator: y[n] = b0*x[n] + a1*y[n-1] + a2*y[n-2]; poles at
258 // r*e^{±j theta}
259 filter.a1 = 2.0f * r * cosf(theta);
260 filter.a2 = -r * r;
261 filter.b0 =
262 1.0f -
263 r; // energy normalization keeping level stable across bandwidths
264 }

267 float applyFormantFilter(FormantFilter& filter, float input) {
268 float y = filter.b0 * input + filter.a1 * filter.y1 + filter.a2 * filter.y2;
269 filter.y2 = filter.y1;
270 filter.y1 = y;
271 return y;
272 }

275 float generateVoiceSource(float t, float f0Base) {
276 // Pitch with jitter updated every few milliseconds
277 const uint16_t jitterIntervalSamples = sampleRate_ / 200; // ~5ms
278 if (++jitterCounter_ >= jitterIntervalSamples) {
279 jitterCounter_ = 0;
280 float jitter = cfg_.jitterPct * frand();
281 // Base the jittered pitch on f0Base (the per-phoneme pitch passed in,
282 // which may carry a stress rise or an explicit pitchHz override from
283 // PhonemeSynthesisParams) rather than the instance's static voiceF0_ --
284 // using voiceF0_ here made f0Base a dead parameter and silently
285 // discarded any per-call pitch adjustment.
286 f0Current_ = f0Base * (1.0f + jitter);
287 }
288 // Add small downward pitch contour over phoneme duration (approx t
289 // fraction)
290 float dynamicF0 = f0Current_ - cfg_.pitchFallPerSec * t;
291 if (dynamicF0 < 50.0f) dynamicF0 = 50.0f;
292 voicePhase_ += 2.0f * M_PI * dynamicF0 / sampleRate_;
293 if (voicePhase_ >= 2.0f * M_PI) voicePhase_ -= 2.0f * M_PI;
294 float phase_norm = voicePhase_ / (2.0f * M_PI);
295 // Simplified LF derivative (two-piece polynomial) + closing spike
296 float glottal = 0.0f;
297 const float openFrac = 0.45f;
298 const float closeFrac = 0.8f;
299 if (phase_norm < openFrac) {
300 float x = phase_norm / openFrac;
301 glottal = x - x * x * x * 0.3333f; // gentle rise
302 } else if (phase_norm < closeFrac) {
303 float x = (phase_norm - openFrac) / (closeFrac - openFrac);
304 glottal = 1.0f - 0.5f * x * x; // quasi-plateau with slight fall
305 } else {
306 float x = (phase_norm - closeFrac) / (1.0f - closeFrac);
307 glottal = 0.2f * (1.0f - x); // abrupt closure & return to zero
308 }
309 // Shimmer (amplitude variation)
310 glottal *= (1.0f + cfg_.shimmerPct * frand());
311 // Spectral tilt
312 spectralTiltState_ = (1.0f - cfg_.spectralTilt) * glottal +
313 cfg_.spectralTilt * spectralTiltState_;
314 // Add controlled breathiness
315 float aspiration = cfg_.breathiness * generateNoiseSource();
316 return 0.55f * spectralTiltState_ + 0.35f * glottal + aspiration;
317 }

324 noiseSeed_ = noiseSeed_ * 1103515245 + 12345;

326 }

363 PhonemeFlags classifyPhoneme(const std::string& p) {
364 PhonemeFlags fl;
365 const PhonemeRule* r = findPhonemeRule(p);
366 if (r) {
367 fl.voiced = (r->flags & PF_VOICED) != 0;
368 fl.strongSibilant =
369 (p == "S" || p == "Z" || p == "s" ||
370 p == "z"); // keep explicit to match prior logic for scaling
371 fl.postalveolar = (r->params.f2 == 2000 &&
372 (p == "SH" || p == "ʃ" || p == "ZH" || p == "ʒ"));
373 fl.nasal = (r->flags & PF_NASAL) != 0;
374 fl.silence = (r->flags & PF_SILENCE) != 0;
375 }
376 return fl;

404 void applyNasalNotchesIfNeeded(float& sample, bool isNasal, size_t i) {
405 if (!(isNasal && cfg_.enableNasalNotch)) return;
406 if (i == 0) {
407 auto configNotch = [&](NotchFilter& nf, float f, float bw) {
408 float r = expf(-M_PI * bw / sampleRate_);
409 float theta = 2.0f * M_PI * f / sampleRate_;
410 nf.b0 = 1.0f;
411 nf.b1 = -2.0f * cosf(theta);
412 nf.b2 = 1.0f;
413 nf.a1 = 2.0f * r * cosf(theta);
414 nf.a2 = -r * r;
415 nf.x1 = nf.x2 = nf.y1 = nf.y2 = 0.0f;
416 nf.enabled = true;
417 };
418 configNotch(nasalNotch1_, cfg_.nasalNotchFreq1, cfg_.nasalNotchBw1);
419 configNotch(nasalNotch2_, cfg_.nasalNotchFreq2, cfg_.nasalNotchBw2);
420 }
421 auto applyNotch = [&](NotchFilter& nf, float in) {
422 if (!nf.enabled) return in;
423 float y = nf.b0 * in + nf.b1 * nf.x1 + nf.b2 * nf.x2 + nf.a1 * nf.y1 +
424 nf.a2 * nf.y2;
425 nf.x2 = nf.x1;
426 nf.x1 = in;
427 nf.y2 = nf.y1;
428 nf.y1 = y;
429 return y;
430 };
431 float before = sample;
432 sample = applyNotch(nasalNotch1_, sample);
433 sample = applyNotch(nasalNotch2_, sample);
434 sample =
435 (1.0f - cfg_.nasalNotchDepth) * before + cfg_.nasalNotchDepth * sample;
436 }

439 void applyDeEsserIfNeeded(float& sample, const PhonemeFlags& flags) {
440 if (flags.voiced) return;
441 if (!(flags.strongSibilant || flags.postalveolar)) return;
442 deEsserLPState_ = deEsserLPState_ + 0.1f * (sample - deEsserLPState_);
443 float high = sample - deEsserLPState_;
444 float ahigh = fabsf(high);
445 if (ahigh > cfg_.deEsserThreshold) {
446 float over =
447 (ahigh - cfg_.deEsserThreshold) / (0.6f - cfg_.deEsserThreshold);
448 if (over > 1.0f) over = 1.0f;
449 float atten = 1.0f - cfg_.deEsserAtten * over;
450 sample *= atten;
451 }
452 }

462 void synthesizeSamples(PhonemeSynthesisContext& ctx) {
463 size_t overlapSamples =
464 static_cast<size_t>((cfg_.overlapMs / 1000.0f) * sampleRate_);
465 if (overlapSamples > ctx.numSamples / 2)
466 overlapSamples = ctx.numSamples / 2;
467 for (size_t i = 0; i < ctx.numSamples; ++i) {
468 float t = (float)i / (float)sampleRate_;
469 float progress =
470 (ctx.numSamples > 1) ? (float)i / (float)(ctx.numSamples - 1) : 0.0f;
471 // Steady-state (or, for diphthongs, mid-phoneme-eased) formant
472 // target for this sample.
473 float tf1, tf2, tf3;
474 if (ctx.dynamic) {
475 float ease = progress < 0.5f
476 ? 2.0f * progress * progress
477 : -1.0f + (4.0f - 2.0f * progress) * progress;
478 tf1 = ctx.params.f1 + (ctx.target2.f1 - ctx.params.f1) * ease;
479 tf2 = ctx.params.f2 + (ctx.target2.f2 - ctx.params.f2) * ease;
480 tf3 = ctx.params.f3 + (ctx.target2.f3 - ctx.params.f3) * ease;
481 } else {
482 tf1 = ctx.params.f1;
483 tf2 = ctx.params.f2;
484 tf3 = ctx.params.f3;
485 }
486 // Coarticulation: for the first attackSamples, blend from where the
487 // PREVIOUS phoneme's formants ended (ctx.entryF1/2/3) toward this
488 // sample's own target, instead of starting already at rest -- real
489 // formant transitions between adjacent phonemes carry much of
490 // speech's intelligibility (especially consonant place-of-
491 // articulation cues), so snapping every phoneme instantly to its own
492 // target made speech sound like separately-glued blobs.
493 if (i < ctx.attackSamples) {
494 float attackProgress = (float)i / (float)ctx.attackSamples;
495 // Raised-cosine ease: smooth start and end of the glide.
496 float ease = 0.5f - 0.5f * cosf((float)M_PI * attackProgress);
497 tf1 = ctx.entryF1 + (tf1 - ctx.entryF1) * ease;
498 tf2 = ctx.entryF2 + (tf2 - ctx.entryF2) * ease;
499 tf3 = ctx.entryF3 + (tf3 - ctx.entryF3) * ease;
500 }
501 lastF1_ = formantSmoothingFactor_ * lastF1_ +
502 (1.0f - formantSmoothingFactor_) * tf1;
503 lastF2_ = formantSmoothingFactor_ * lastF2_ +
504 (1.0f - formantSmoothingFactor_) * tf2;
505 lastF3_ = formantSmoothingFactor_ * lastF3_ +
506 (1.0f - formantSmoothingFactor_) * tf3;
507 if ((i & 31) == 0) {
508 configureFormantFilter(f1Filter_, lastF1_, ctx.params.bw1);
509 configureFormantFilter(f2Filter_, lastF2_, ctx.params.bw2);
510 configureFormantFilter(f3Filter_, lastF3_, ctx.params.bw3);
511 }
512 float source = 0.0f;
513 if (ctx.flags.voiced) {
514 float f0Adj = ctx.f0;
515 if (ctx.stressPitchRise > 0.0f) {
516 float sd = progress / (cfg_.stressDecayPortion <= 0.01f
517 ? 0.01f
518 : cfg_.stressDecayPortion);
519 if (sd > 1.0f) sd = 1.0f;
520 float decay = expf(-2.2f * sd);
521 f0Adj += ctx.stressPitchRise * decay;
522 }
523 float g = generateVoiceSource(progress * ctx.duration, f0Adj);
524 float harmonics = 0.0f;
525 const int NH = 5;
526 for (int h = 2; h <= NH; ++h)
527 harmonics += (1.0f / (float)h) * sinf(voicePhase_ * h);
528 float voicedSource = 0.65f * g + 0.30f * harmonics;
529 if (ctx.voicing >= 1.0f) {
530 source = voicedSource;
531 } else {
532 // Blend toward an unvoiced/whispered version of this phoneme:
533 // replace the harmonic glottal source with broadband noise as
534 // voicing decreases from 1.0 (fully voiced) to 0.0 (whisper).
535 // Naturally unvoiced phonemes never reach this branch, so this
536 // only affects vowels/voiced consonants.
537 float n = generateNoiseSource();
538 lp_hp_ = 0.85f * lp_hp_ + 0.15f * n;
539 float hp = n - lp_hp_;
540 float clampedVoicing = ctx.voicing < 0.0f ? 0.0f : ctx.voicing;
541 source = voicedSource * clampedVoicing + hp * (1.0f - clampedVoicing);
542 }
543 } else {
544 float n = generateNoiseSource();
545 lp_hp_ = 0.85f * lp_hp_ + 0.15f * n;
546 float hp = n - lp_hp_;
547 if (ctx.flags.strongSibilant) {
548 float bp = hp - 0.6f * prevNoise1_;
549 prevNoise1_ = hp;
550 source = cfg_.sibilantGain * bp;
551 } else if (ctx.flags.postalveolar) {
552 float bp = (hp - prevNoise2_);
553 prevNoise2_ = hp;
554 source = cfg_.postalveolarGain * bp;
555 } else
556 source = hp;
557 }
558 float out1 = applyFormantFilter(f1Filter_, source) * ctx.params.a1;
559 float out2 = applyFormantFilter(f2Filter_, source) * ctx.params.a2;
560 float out3 = applyFormantFilter(f3Filter_, source) * ctx.params.a3;
561 float out4 = 0.0f;
562 if (cfg_.enableF4) out4 = applyFormantFilter(f4Filter_, source) * 0.25f;
563 if (!ctx.flags.voiced && ctx.flags.strongSibilant)
564 out3 *= cfg_.sibilantHFFormantScale;
565 float sample = out1 + out2 + out3 + out4;
566 applyNasalNotchesIfNeeded(sample, ctx.flags.nasal, i);
567 sample = 0.9f * sample + 0.1f * prevSample_;
568 prevSample_ = sample;
569 float envelope = generateEnvelope(t, ctx.duration);
570 sample *= envelope;
571 finalizeSampleNasalLP(sample, ctx.flags.nasal);
572 applyDeEsserIfNeeded(sample, ctx.flags);
573 if (sample > 1.0f) sample = 1.0f - 0.5f * (sample - 1.0f);
574 if (sample < -1.0f) sample = -1.0f - 0.5f * (sample + 1.0f);
575 audioBuffer_[i] = (int16_t)(sample * 15000.0f);
576 }
577 }

580 void applyRMSNormalization(size_t numSamples, const PhonemeFlags& flags) {
581 int64_t sumSq = 0;
582 for (size_t i = 0; i < numSamples; ++i) {
583 int32_t v = audioBuffer_[i];
584 sumSq += (int64_t)v * v;
585 }
586 float rms =
587 (numSamples > 0) ? sqrtf((float)sumSq / (float)numSamples) : 0.0f;
588 float target = flags.voiced ? cfg_.targetRMS
589 : ((flags.strongSibilant || flags.postalveolar)
590 ? cfg_.fricativeTargetRMS
591 : cfg_.targetRMS * 0.9f);
592 if (rms > 1.0f) {
593 float desiredGain = target / rms;
594 if (desiredGain > 1.5f) desiredGain = 1.5f;
595 adaptiveGain_ = (1.0f - cfg_.rmsAdaptRate) * adaptiveGain_ +
596 cfg_.rmsAdaptRate * desiredGain;
597 for (size_t i = 0; i < numSamples; ++i) {
598 int32_t v = (int32_t)(audioBuffer_[i] * adaptiveGain_);
599 if (v > 32767)
600 v = 32767;
601 else if (v < -32768)
602 v = -32768;
603 audioBuffer_[i] = (int16_t)v;
604 }
605 }

609 void applySibilantPostProcessing(size_t numSamples,
610 const PhonemeFlags& flags) {
611 if (flags.voiced || !(flags.strongSibilant || flags.postalveolar)) return;
612 // Peak limiting
613 int16_t peak = 0;
614 for (size_t i = 0; i < numSamples; ++i) {
615 int16_t v = audioBuffer_[i];
616 int16_t a = v < 0 ? -v : v;
617 if (a > peak) peak = a;
618 }
619 float targetP = cfg_.fricativeMaxPeak;
620 if (peak > targetP && peak > 0) {
621 float scale = targetP / (float)peak;
622 for (size_t i = 0; i < numSamples; ++i) {
623 audioBuffer_[i] = (int16_t)(audioBuffer_[i] * scale);
624 }
625 }
626 // Dynamic high-band compression
627 float lp = 0.0f;
628 float sumSqHB = 0.0f;
629 for (size_t i = 0; i < numSamples; ++i) {
630 float x = (float)audioBuffer_[i];
631 lp = lp + 0.08f * (x - lp);
632 float hb = x - lp;
633 sumSqHB += hb * hb;
634 }
635 float rmsHB =
636 (numSamples > 0) ? sqrtf(sumSqHB / numSamples) : 0.0f;
637 float targetHB = cfg_.sibilantDynamicTargetRMS;
638 if (rmsHB > targetHB && rmsHB > 1.0f) {
639 float over = rmsHB / targetHB;
640 float expn = (1.0f / cfg_.sibilantCompressionRatio) - 1.0f;
641 float desiredGain = powf(over, expn);
642 if (desiredGain < 0.25f) desiredGain = 0.25f;
643 float smooth = (desiredGain < sibilantDynGain_) ? cfg_.sibilantAttack
644 : cfg_.sibilantRelease;
645 sibilantDynGain_ =
646 smooth * sibilantDynGain_ + (1.0f - smooth) * desiredGain;
647 for (size_t i = 0; i < numSamples; ++i) {
648 int32_t v = (int32_t)(audioBuffer_[i] * sibilantDynGain_);
649 if (v > 32767)
650 v = 32767;
651 else if (v < -32768)
652 v = -32768;
653 audioBuffer_[i] = (int16_t)v;
654 }
655 } else {
656 sibilantDynGain_ = cfg_.sibilantRelease * sibilantDynGain_ +
657 (1.0f - cfg_.sibilantRelease);
658 }
659 if (cfg_.sibilantLowpassMix > 0.0f) {
660 float lp2 = 0.0f;
661 float mix = cfg_.sibilantLowpassMix;
662 for (size_t i = 0; i < numSamples; ++i) {
663 float x = (float)audioBuffer_[i];
664 lp2 += 0.12f * (x - lp2);
665 float softened = lp2;
666 float y = x * (1.0f - mix) + softened * mix;
667 int32_t v = (int32_t)(y);
668 if (v > 32767)
669 v = 32767;
670 else if (v < -32768)
671 v = -32768;
672 audioBuffer_[i] = (int16_t)v;
673 }
674 }
675 if (cfg_.sibilantFinalGain < 0.999f) {
676 float g = cfg_.sibilantFinalGain;
677 for (size_t i = 0; i < numSamples; ++i) {
678 int32_t v = (int32_t)(audioBuffer_[i] * g);
679 if (v > 32767)
680 v = 32767;
681 else if (v < -32768)
682 v = -32768;
683 audioBuffer_[i] = (int16_t)v;
684 }
685 }
686 }

690 void applyVolumeFactor(size_t numSamples, float callVolume = 1.0f) {
691 float gain = volumeFactor_ * callVolume;
692 for (size_t i = 0; i < numSamples; ++i) {
693 int32_t v = (int32_t)(gain * audioBuffer_[i]);
694 if (v > 32767)
695 v = 32767;
696 else if (v < -32768)
697 v = -32768;
698 audioBuffer_[i] = (int16_t)v;
699 }
700 }
701
703 void applyCrossfade(size_t numSamples) {
704 if (!previousTail_.empty()) {
705 size_t overlap = std::min(previousTail_.size(), numSamples);
706 for (size_t i = 0; i < overlap; ++i) {
707 float w = (float)i / (float)(overlap - 1);
708 float a = previousTail_[i] * (1.0f - w);
709 float b = audioBuffer_[i] * w;
710 int32_t mix = (int32_t)a + (int32_t)b;
711 if (mix > 32767) mix = 32767;
712 if (mix < -32768) mix = -32768;
713 audioBuffer_[i] = (int16_t)mix;
714 }
715 previousTail_.clear();
716 }
717 size_t overlapSamples =
718 static_cast<size_t>((cfg_.overlapMs / 1000.0f) * sampleRate_);
719 if (overlapSamples > 8 && overlapSamples < numSamples) {
720 // Bound the capture to the samples actually written for THIS phoneme
721 // (numSamples), not audioBuffer_.end() -- audioBuffer_ never shrinks,
722 // so its size() can exceed numSamples whenever an earlier phoneme was
723 // longer, and audioBuffer_.end() would then point past what was just
724 // written, capturing stale leftover data as the crossfade tail.
725 previousTail_.assign(audioBuffer_.begin() + (numSamples - overlapSamples),
726 audioBuffer_.begin() + numSamples);
727 }
728 }

760 void preparePhonemeSynthesis(const std::string& phoneme,
761 PhonemeSynthesisContext& ctx,
762 const PhonemeSynthesisParams& params) {
763 ctx.phonStr = stripStress(phoneme, ctx.stress);
764 float defaultDuration = getPhoneDuration(ctx.phonStr);
765 ctx.duration = params.durationMs > 0
766 ? (params.durationMs / 1000.0f)
767 : defaultDuration;
768 ctx.numSamples = (size_t)(ctx.duration * sampleRate_);
769 if (ctx.numSamples > audioBuffer_.size())
770 audioBuffer_.resize(ctx.numSamples);
771 ctx.params = getFormantParams(ctx.phonStr);
772 ctx.target2 = ctx.params;
773 ctx.dynamic = false;
774 setupDiphthongTargets(ctx.phonStr, ctx.target2, ctx.dynamic);
775 randomizeFormantsIfNeeded(ctx.params, ctx.phonStr);
776 configureInitialFilters(ctx.params);
777 // Snapshot where the previous phoneme's formants ended up (see the
778 // coarticulation comment on PhonemeSynthesisContext), and size an
779 // attack window to glide from there to this phoneme's own target: up
780 // to ~30ms, capped at 30% of this phoneme's own duration so very short
781 // stops don't spend their whole duration transitioning.
782 ctx.entryF1 = lastF1_;
783 ctx.entryF2 = lastF2_;
784 ctx.entryF3 = lastF3_;
785 float attackSec = 0.03f < ctx.duration * 0.3f ? 0.03f : ctx.duration * 0.3f;
786 ctx.attackSamples = (size_t)(attackSec * sampleRate_);
787 if (ctx.attackSamples < 1) ctx.attackSamples = 1;
788 ctx.flags = classifyPhoneme(ctx.phonStr);
789 ctx.f0 = params.pitchHz > 0.0f ? params.pitchHz : voiceF0_;
790 if (ctx.params.f1 < 350.0f)
791 ctx.f0 *= 1.05f; // slight boost for closed vowels
792 ctx.stressPitchRise = computeStressPitchRise(ctx.stress);
793 ctx.voicing = params.voicing;
794 }

797 bool handleSilenceSetup(PhonemeSynthesisContext& ctx, ::Print& out) {
798 if (!ctx.flags.silence) return false;
799
800 size_t numSamples = ctx.numSamples;
801 if (numSamples == 0) return true;
802
803 // Determine overlap samples (mirror logic from applyCrossfade)
804 size_t overlapSamples = static_cast<size_t>((cfg_.overlapMs / 1000.0f) * sampleRate_);
805 if (overlapSamples > 8 && overlapSamples < numSamples) {
806 // we'll set previousTail_ at end to zeros of this size
807 } else {
808 overlapSamples = 0; // treat as no tail storage
809 }
810
811 // 1. If there is a previous tail, perform crossfade (fade it out to zero)
812 if (!previousTail_.empty()) {
813 size_t overlap = std::min(previousTail_.size(), numSamples);
814 if (overlap > 0) {
815 // Create a small temporary buffer just for the overlap mix
816 std::vector<int16_t> temp(overlap);
817 for (size_t i = 0; i < overlap; ++i) {
818 float w = (float)i / (float)(overlap - 1);
819 // previous contribution fades out; silence fades in (0)
820 float a = previousTail_[i] * (1.0f - w);
821 int32_t v = (int32_t)a;
822 if (v > 32767) v = 32767; else if (v < -32768) v = -32768;
823 temp[i] = (int16_t)v;
824 }
825 out.write(reinterpret_cast<const uint8_t*>(temp.data()), overlap * sizeof(int16_t));
826 numSamples -= overlap;
827 }
828 previousTail_.clear();
829 }
830
831 // 2. Stream the remaining silence as zero chunks without allocating a giant buffer
832 static int16_t zeroChunk[256] = {0};
833 while (numSamples > 0) {
834 size_t n = (numSamples > 256) ? 256 : numSamples;
835 out.write(reinterpret_cast<const uint8_t*>(zeroChunk), n * sizeof(int16_t));
836 numSamples -= n;
837 }
838
839 // 3. Prepare a zero tail for next phoneme if overlap policy applies
840 if (overlapSamples > 0) {
841 previousTail_.assign(overlapSamples, 0);
842 }
843 return true;
844 }

867 float generateEnvelope(float t, float duration) {
868 float progress = t / duration;
869
870 // Different envelopes for different phoneme types
871 if (duration < 0.08f) {
872 // Short phonemes (consonants) - quick attack/decay
873 if (progress < 0.3f) {
874 return progress / 0.3f; // Quick attack
875 } else if (progress > 0.7f) {
876 return (1.0f - progress) / 0.3f; // Quick decay
877 } else {
878 return 1.0f; // Brief sustain
879 }
880 } else {
881 // Longer phonemes (vowels) - more gradual envelope
882 float attack = 0.15f;
883 float release = 0.25f;
884
885 if (progress < attack) {
886 // Smooth attack with exponential curve
887 float attack_progress = progress / attack;
888 return 1.0f - expf(-5.0f * attack_progress);
889 } else if (progress > (1.0f - release)) {
890 // Smooth release with exponential decay
891 float release_progress = (progress - (1.0f - release)) / release;
892 return expf(-3.0f * release_progress);
893 } else {
894 // Sustain with slight amplitude variation
895 float sustain_variation = 1.0f + 0.05f * sinf(2.0f * (float)M_PI * t * 4.0f);
896 return 0.95f * sustain_variation;
897 }
898 }
