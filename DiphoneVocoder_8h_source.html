128 bool processSequenceWithLookahead(const std::string& sequence, ::Print& out,
129 const PhonemeSynthesisParams& params) override {
130 if (!isReady()) {
131 return false;
132 }
133
134 // Parse phoneme sequence
135 std::vector<std::string> phonemes = StringUtils::split(sequence, ' ');
136 if (phonemes.empty()) {
137 return false;
138 }
139
140 // Strip stress digits (e.g. "IH1" -> "IH") before building diphone
141 // names or looking up durations: diphone names are keyed by bare
142 // phoneme symbols, but a G2P source using the full CMU dictionary
143 // (COMPACT_CMUDICT_EN) emits stress-marked phonemes -- without this,
144 // any word with a stressed vowel would fail every diphone lookup.
145 for (std::string& phoneme : phonemes) {
146 int stress;
147 phoneme = stripStressMarker(phoneme, stress);
148 }
149
150 // AH0 and ER0 are NOT stress-marked variants of AH/ER (stripStressMarker
151 // correctly leaves them alone -- they're genuinely distinct reduced-
152 // vowel phonemes in Phonemes.h's phoneme_map). But the diphone corpus
153 // (generate_relevant_diphones.sh) was only ever generated for the 15
154 // main vowels -- it has no AH0/ER0 diphone data at all (e.g. no
155 // "P ER0"), so any word using them (e.g. "whispers" -> "... P ER0 Z")
156 // silently failed every diphone lookup touching that vowel, dropping
157 // that part of the word. Fall back to the closest available full-vowel
158 // unit rather than dropping audio.
159 for (std::string& phoneme : phonemes) {
160 if (phoneme == "AH0") phoneme = "AH";
161 else if (phoneme == "ER0") phoneme = "ER";
162 }
163
164 // A whole utterance arrives as ONE sequence with silence-class tokens
165 // (SIL/SP, inserted by TinyTTSTools::toPhonemes() between words) mixed
166 // in. Those aren't phonemes with their own diphone entries -- attempting
167 // e.g. "N SP" or "SP F" as a diphone name always fails -- so split the
168 // sequence into silence-free segments (typically one word each),
169 // synthesize each as its own SIL-bracketed diphone chain, and render
170 // each silence-class token as real silence of its own duration in
171 // between. A single phoneme with no silence anywhere in the sequence
172 // (isReady()-checked callers may pass just one word) is handled the
173 // same way, as a one-segment sequence.
174 PhonemeType phonemeType = dictionary_.phonemeType();
175 bool success = true;
176 std::vector<std::string> segment;
177 auto flushSegment = [&]() {
178 if (segment.empty()) return;
179 if (!synthesizeSegment(segment, out, params, phonemeType)) success = false;
180 segment.clear();
181 };
182 for (const std::string& token : phonemes) {
183 if (phonemes_.isSilence(phonemeType, token.c_str())) {
184 flushSegment();
185 uint16_t duration = resolveDuration(phonemes_.getPhonemeDuration(phonemeType, token), params);
186 outputSilence(sampleRate(), duration, out);
187 } else {
188 segment.push_back(token);
189 }
190 }
191 flushSegment();
192
193 return success;
194 }

204 bool synthesizeSegment(std::vector<std::string> phonemes, ::Print& out,
205 const PhonemeSynthesisParams& params, PhonemeType phonemeType) {
206 // A diphone audio unit (source file "X_Y.wav", looked up as "X Y" --
207 // see SoundEntry.h on why the lookup key uses a space) only ever
208 // supplies the SECOND half of X and the FIRST half of Y (see
209 // generate_relevant_diphones.sh). Sliding over adjacent pairs within a
210 // segment therefore never renders the very first phoneme's own onset
211 // (its first half, from silence) or the very last phoneme's own
212 // release (its second half, into silence) -- the segment would start
213 // mid-attack and end mid-release. Bracket it with SIL on both ends so
214 // the leading "SIL <first>" and trailing "<last> SIL" diphones (both
215 // present in the corpus -- see Data/wav/diphones/) get generated too.
216 if (phonemes.front() != "SIL") phonemes.insert(phonemes.begin(), "SIL");
217 if (phonemes.back() != "SIL") phonemes.push_back("SIL");
218
219 bool success = true;
220
221 // Slide over every ADJACENT pair (i, i+1), not disjoint pairs (0,1),
222 // (2,3), ... -- N phonemes have N-1 real transitions between them, and
223 // each one needs its own diphone. Stepping by 2 instead of 1 silently
224 // skipped every other transition (e.g. "HH EH L OW" only ever
225 // synthesized HH_EH and L_OW, never the EH->L transition in the
226 // middle), replacing real coarticulated diphone audio with a generic
227 // cross-fade between unrelated diphone units for half of any utterance.
228 // Diphone audio is generated as roughly half-of-phone1 + half-of-phone2
229 // (see setup/audio/diphones/generate_relevant_diphones.sh), so playing
230 // consecutive diphones back-to-back reconstructs each shared phoneme's
231 // duration once, not twice.
232 for (size_t i = 0; i + 1 < phonemes.size(); i++) {
233 std::string diphoneName = phonemes[i] + " " + phonemes[i + 1];
234
235 // Calculate duration for the diphone: each side contributes roughly
236 // half its natural phoneme duration, matching how the audio itself
237 // was generated -- except SIL, a special case (see
238 // diphoneSideDurationMs()'s doc).
239 uint16_t duration1 = diphoneSideDurationMs(phonemeType, phonemes[i]);
240 uint16_t duration2 = diphoneSideDurationMs(phonemeType, phonemes[i + 1]);
241 uint16_t diphoneDuration = resolveDuration(duration1 + duration2, params);
242
243 // Get current diphone entry
244 const SoundEntry* currentEntry = getAudioEntry(diphoneName);
245 if (!currentEntry) {
246 success = false;
247 continue;
248 }
249
250 // Get next diphone entry for enhanced concatenation
251 const SoundEntry* nextEntry = nullptr;
252 if (i + 2 < phonemes.size()) {
253 std::string nextDiphoneName = phonemes[i + 1] + " " + phonemes[i + 2];
254 nextEntry = getNextAudioEntry(diphoneName, nextDiphoneName);
255 }
256
257 // Process with enhanced concatenation
258 processAudioUnit(currentEntry, nextEntry, diphoneDuration, out, params);
259 }
260
261 return success;
262 }
263
264 // The fixed silence portion generate_relevant_diphones.sh actually bakes
265 // into every SIL_<phoneme>/<phoneme>_SIL recording's boundary side (its
266 // own SILENCE_BOUNDARY_MS constant) -- unrelated to SIL's own
267 // Phonemes.h table duration (400ms), which is how long a real inter-word
268 // *pause* should last, not how much silence a diphone recording's edge
269 // contains. Confirmed empirically: real SIL_HH/HH_SIL/SIL_T/T_SIL
270 // recordings measure ~87.5-128ms total, matching this constant's
271 // contribution + the other side's real half-duration; using SIL's own
272 // 400ms/2=200ms here instead overstated every boundary diphone's target
273 // by 100-160ms. That happened to have no audible effect only because
274 // processUnitCombination() treats the duration as a truncation ceiling,
275 // never a stretch/pad target -- an inflated-but-unenforced number is
276 // still wrong to compute, and a future change to that consumption logic
277 // could turn this into real dead-air at every word boundary.
278 static constexpr uint16_t kBoundarySilenceMs = 50;
279
284 uint16_t diphoneSideDurationMs(PhonemeType phonemeType, const std::string& phoneme) const {
285 if (phoneme == "SIL") return kBoundarySilenceMs;
286 return phonemes_.getPhonemeDuration(phonemeType, phoneme) / 2;
287 }
