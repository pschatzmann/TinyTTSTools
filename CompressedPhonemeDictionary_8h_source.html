86 bool lookup(const std::string& word, std::string& outPhonemes) const override {
87 if (count_ == 0) return false;
88 int32_t lo = 0, hi = static_cast<int32_t>(count_) - 1;
89 while (lo <= hi) {
90 int32_t mid = lo + (hi - lo) / 2;
91 uint32_t start = wordByteOffset(static_cast<size_t>(mid));
92 uint8_t len = wordLengths_[mid];
93 int cmp = compareWordAt(start, len, word);
94 if (cmp == 0) {
95 decodePhonemesAt(static_cast<size_t>(mid), outPhonemes);
96 return true;
97 } else if (cmp < 0) {
98 lo = mid + 1;
99 } else {
100 hi = mid - 1;
101 }
102 }
103 return false;
104 }

147 struct DecodeTables {
148 static constexpr uint8_t kMaxLen = 16;
149 uint32_t firstCode[kMaxLen + 1] = {};
150 uint8_t countByLen[kMaxLen + 1] = {};
151 uint8_t symbolsByLen[kMaxLen + 1][PHONEME_HUFFMAN_CODE_COUNT] = {};
152 uint8_t maxLen = 0;
153 };
154
155 static const DecodeTables& decodeTables() {
156 static const DecodeTables t = [] {
157 DecodeTables d;
158 for (size_t i = 0; i < PHONEME_HUFFMAN_CODE_COUNT; ++i) {
159 const auto& e = PHONEME_HUFFMAN_CODES[i];
160 if (d.countByLen[e.length] == 0) d.firstCode[e.length] = e.code;
161 d.symbolsByLen[e.length][d.countByLen[e.length]] = e.packedByte;
162 d.countByLen[e.length]++;
163 if (e.length > d.maxLen) d.maxLen = e.length;
164 }
165 return d;
166 }();
167 return t;
168 }

170 uint8_t getBit(uint32_t bitIndex) const {
171 uint32_t byteIdx = bitIndex >> 3;
172 uint8_t bitPos = 7 - static_cast<uint8_t>(bitIndex & 7);
173 return (phonemeBits_[byteIdx] >> bitPos) & 1;
174 }

177 uint8_t decodeOneSymbol(uint32_t bitOffset, uint8_t& symOut) const {
178 const DecodeTables& t = decodeTables();
179 uint32_t code = 0;
180 for (uint8_t len = 1; len <= t.maxLen; ++len) {
181 code = (code << 1) | getBit(bitOffset + len - 1);
182 uint8_t cnt = t.countByLen[len];
183 if (cnt > 0 && code >= t.firstCode[len] && (code - t.firstCode[len]) < cnt) {
184 symOut = t.symbolsByLen[len][code - t.firstCode[len]];
185 return len;
186 }
187 }
188 symOut = 0;
189 return 1; // malformed stream -- shouldn't happen with valid data
190 }

196 uint8_t cnt = phonemeCounts_[i];

199 bitOff += decodeOneSymbol(bitOff, sym);
200 }
201 }

205 void decodePhonemesAt(size_t idx, std::string& out) const {
206 out.clear();
207 uint32_t bitOff = phonemeBitOffset(idx);
208 uint8_t cnt = phonemeCounts_[idx];
209 static const Phonemes phonemeTable;
210 for (uint8_t k = 0; k < cnt; ++k) {
211 uint8_t sym;
212 bitOff += decodeOneSymbol(bitOff, sym);
213 uint8_t id = sym >> 2;
214 uint8_t stress = sym & 0x3;
215 const PhonemeInfo* info = phonemeTable.getPhonemeById(id);
216 if (!info) continue;
217 if (!out.empty()) out += ' ';
218 out += info->arpabet;
219 if (stress != 0) out += static_cast<char>('0' + stress);
220 }
221 }
