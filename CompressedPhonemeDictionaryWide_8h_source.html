108 bool lookup(const std::string& word, std::string& outPhonemes) const override {
109 if (count_ == 0) return false;
110 int32_t lo = 0, hi = static_cast<int32_t>(count_) - 1;
111 while (lo <= hi) {
112 int32_t mid = lo + (hi - lo) / 2;
113 uint32_t start = wordByteOffset(static_cast<size_t>(mid));
114 uint8_t len = wordLengths_[mid];
115 int cmp = compareWordAt(start, len, word);
116 if (cmp == 0) {
117 decodePhonemesAt(static_cast<size_t>(mid), outPhonemes);
118 return true;
119 } else if (cmp < 0) {
120 lo = mid + 1;
121 } else {
122 hi = mid - 1;
123 }
124 }
125 return false;
126 }

165 void buildDecodeTables(const PhonemeHuffmanCodeWide* codes, size_t codeCount) {
166 decodeTables_ = DecodeTables{};
167 for (size_t i = 0; i < codeCount; ++i) {
168 const auto& e = codes[i];
169 if (e.length > DecodeTables::kMaxLen) continue; // shouldn't happen; see generator
170 if (decodeTables_.countByLen[e.length] == 0) decodeTables_.firstCode[e.length] = e.code;
171 decodeTables_.symbolsByLen[e.length].push_back(e.packedSymbol);
172 decodeTables_.countByLen[e.length]++;
173 if (e.length > decodeTables_.maxLen) decodeTables_.maxLen = e.length;
174 }
175 }

184 int compareWordAt(uint32_t start, uint8_t len, const std::string& word) const {
185 const char* w = wordsBlob_ + start;
186 size_t minLen = static_cast<size_t>(len) < word.size() ? len : word.size();
187 int cmp = std::memcmp(w, word.data(), minLen);
188 if (cmp != 0) return cmp;
189 return static_cast<int>(len) - static_cast<int>(word.size());

199 uint8_t decodeOneSymbol(uint32_t bitOffset, uint16_t& symOut) const {
200 uint32_t code = 0;
201 for (uint8_t len = 1; len <= decodeTables_.maxLen; ++len) {
202 code = (code << 1) | getBit(bitOffset + len - 1);
203 uint8_t cnt = decodeTables_.countByLen[len];
204 if (cnt > 0 && code >= decodeTables_.firstCode[len] &&
205 (code - decodeTables_.firstCode[len]) < cnt) {
206 symOut = decodeTables_.symbolsByLen[len][code - decodeTables_.firstCode[len]];
207 return len;
208 }
209 }
210 symOut = 0;
211 return 1; // malformed stream -- shouldn't happen with valid data
212 }

214 uint32_t phonemeBitOffset(size_t idx) const {
215 size_t block = idx / BLOCK_SIZE;
216 uint32_t bitOff = phonemeBlockBitOffsets_[block];
217 for (size_t i = block * BLOCK_SIZE; i < idx; ++i) {
218 uint8_t cnt = phonemeCounts_[i];
219 for (uint8_t k = 0; k < cnt; ++k) {
220 uint16_t sym;
221 bitOff += decodeOneSymbol(bitOff, sym);
222 }
223 }
224 return bitOff;
225 }

227 void decodePhonemesAt(size_t idx, std::string& out) const {
228 out.clear();
229 uint32_t bitOff = phonemeBitOffset(idx);
230 uint8_t cnt = phonemeCounts_[idx];
231 static const Phonemes phonemeTable;
232 for (uint8_t k = 0; k < cnt; ++k) {
233 uint16_t sym;
234 bitOff += decodeOneSymbol(bitOff, sym);
235 uint8_t id = (sym >> 2) & 0x7F;
236 uint8_t stress = sym & 0x3;
237 PhonemeModifier modifier = static_cast<PhonemeModifier>((sym >> 9) & 0x1F);
238 const PhonemeInfo* info = phonemeTable.getPhonemeById(id);
239 if (!info) continue;
240 if (!out.empty()) out += ' ';
241 std::string token = info->arpabet;
242 if (stress != 0) token += static_cast<char>('0' + stress);
243 else if (modifier != PhonemeModifier::MOD_NONE) token = applyModifierTag(token, modifier);
244 out += token;
245 }
